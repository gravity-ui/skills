                и состояние оплаты.
              </p>
            </div>
            <button className=\"button primary\" onClick={() => setPopover(null)}>
              Всё понятно
            </button>
          </div>
        </div>
      )}
      {toast && (
        <div className=\"toast\" role=\"status\">
          <span>
            <Icon name=\"check\" size={16} />
          </span>
          {toast}
          <button
            className=\"icon-button\"
            onClick={() => setToast('')}
            aria-label=\"Закрыть уведомление\"
          >
            <Icon name=\"close\" size={15} />
          </button>
        </div>
      )}
    </div>
  );
}
