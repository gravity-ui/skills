                          ].map((v) => ({ value: v, content: v })),
                        ]}
                      />
                      <label>Статус</label>
                      <Select
                        width=\"max\"
                        value={[status]}
                        onUpdate={(v) => {
                          setStatus(v[0]);
                          setPage(1);
                        }}
                        options={[
                          { value: \"all\", content: \"Все статусы\" },
                          ...[\"Активен\", \"В отпуске\", \"Неактивен\"].map((v) => ({
                            value: v,
                            content: v,
                          })),
                        ]}
                      />
                      <Button
                        view=\"flat-secondary\"
                        width=\"max\"
                        onClick={() => {
                          setDepartment(\"all\");
                          setStatus(\"all\");
                          setPage(1);
                        }}
                      >
                        Сбросить фильтры
                      </Button>
                    </div>
                  )}
                </div>
                <Button view=\"outlined\" size=\"l\" onClick={exportData}>
                  <Icon name=\"download\" size={17} />
                  Экспорт<span className=\"export-label\">CSV</span>
                </Button>
              </div>
            </div>
            {selected.size > 0 && (
              <div className=\"selection-bar\">
                <span>
                  <strong>Выбрано: {selected.size}</strong>
                  <button onClick={() => setSelected(new Set())}>
                    Снять выделение
                  </button>
                </span>
                <button onClick={exportData}>
                  <Icon name=\"download\" size={15} />
                  Экспорт выбранных
                </button>
              </div>
            )}
            <div className=\"table-scroll\">
              <table>
                <thead>
                  <tr>
                    <th className=\"check-cell\">
                      <Checkbox
                        checked={allChecked}
                        indeterminate={!allChecked && someChecked}
                        onUpdate={togglePage}
                        controlProps={{ \"aria-label\": \"Выбрать все строки на странице\" }}
                      />
                    </th>
                    {(
                      [
                        [\"name\", \"Сотрудник\"],
                        [\"role\", \"Должность\"],
                        [\"department\", \"Отдел\"],
                        [\"status\", \"Статус\"],
                        [\"date\", \"Дата добавления\"],
                      ] as [SortKey, string][]
                    ).map(([key, label]) => (
                      <th
                        key={key}
                        aria-sort={
                          sortActive && sort.key === key
                            ? sort.asc
                              ? \"ascending\"
                              : \"descending\"
                            : \"none\"
                        }
                      >
                        <button
                          className={
                            \"sort-button \" +
                            (sortActive && sort.key === key ? \"sorted\" : \"\")
                          }
                          onClick={() => changeSort(key)}
                        >
                          {label}
                          <span
                            className={
                              sortActive && sort.key === key && sort.asc
                                ? \"sort-icon ascending\"
                                : \"sort-icon\"
                            }
                          >
                            <Icon name=\"arrow\" size={14} />
                          </span>
                        </button>
                      </th>
                    ))}
                    <th className=\"row-menu-cell\" />
                  </tr>
                </thead>
                <tbody>
                  {visible.map((p) => (
                    <tr
                      key={p.id}
                      className={selected.has(p.id) ? \"selected-row\" : \"\"}
                    >
                      <td className=\"check-cell\">
                        <Checkbox
                          checked={selected.has(p.id)}
                          onUpdate={() => toggle(p.id)}
                          controlProps={{ \"aria-label\": `Выбрать: ${p.name}` }}
                        />
                      </td>
                      <td>
                        <button
                          className=\"person-cell\"
                          onClick={() => setDetail(p)}
                        >
                          <span className={`avatar avatar-${p.color}`}>
                            {initials(p.name)}
                          </span>
                          <span>
                            <strong>{p.name}</strong>
                            <small>{p.email}</small>
                          </span>
                        </button>
                      </td>
                      <td>
                        <span className=\"role-text\">{p.role}</span>
                      </td>
                      <td>
                        <span className={\"department-tag dept-\" + p.department}>
                          {p.department}
                        </span>
                      </td>
                      <td>
                        <span
                          className={
                            \"status-tag \" +
                            (p.status === \"Активен\"
                              ? \"status-active\"
                              : p.status === \"В отпуске\"
                                ? \"status-away\"
                                : \"status-inactive\")
                          }
                        >
                          <i />
                          {p.status}
                        </span>
                      </td>
                      <td className=\"date-cell\">{dateLabel(p.date)}</td>
                      <td className=\"row-menu-cell\">
                        <button
                          className=\"row-more\"
                          aria-label={`Подробнее: ${p.name}`}
                          onClick={() => setDetail(p)}
                        >
                          ···
                        </button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
              {visible.length === 0 && (
                <div className=\"empty-state\">
                  <Icon name=\"search\" size={30} />
                  <h3>Сотрудники не найдены</h3>
                  <p>Попробуйте другой запрос или сбросьте фильтры.</p>
                  <Button
                    view=\"outlined\"
                    onClick={() => {
                      setQuery(\"\");
                      setDepartment(\"all\");
                      setStatus(\"all\");
                      setTab(\"Все сотрудники\");
                    }}
                  >
                    Сбросить всё
                  </Button>
                </div>
              )}
            </div>
            <div className=\"pagination\">
              <span className=\"results-text\">
                Показано{\" \"}
                <strong>
                  {filtered.length ? (currentPage - 1) * pageSize + 1 : 0}–
                  {Math.min(currentPage * pageSize, filtered.length)}
                </strong>{\" \"}
                из <strong>{filtered.length}</strong> сотрудников
              </span>
              <div className=\"pagination-controls\">
                <span className=\"per-page-label\">На странице</span>
                <Select
                  size=\"m\"
                  value={[String(pageSize)]}
                  options={[8, 16, 24].map((n) => ({
                    value: String(n),
                    content: String(n),
                  }))}
                  onUpdate={(v) => {
                    setPageSize(Number(v[0]));
                    setPage(1);
                  }}
                  aria-label=\"Количество строк на странице\"
                />
                <div className=\"page-buttons\">
                  <button
                    aria-label=\"Предыдущая страница\"
                    disabled={currentPage === 1}
                    onClick={() => setPage(currentPage - 1)}
                  >
                    <span className=\"previous\">
                      <Icon name=\"chevron\" size={16} />
                    </span>
                  </button>
                  {Array.from({ length: pages }, (_, i) => i + 1).map((n) => (
                    <button
                      key={n}
                      className={currentPage === n ? \"page-active\" : \"\"}
                      aria-label={`Страница ${n}`}
                      aria-current={currentPage === n ? \"page\" : undefined}
                      onClick={() => setPage(n)}
                    >
                      {n}
                    </button>
                  ))}
                  <button
                    aria-label=\"Следующая страница\"
                    disabled={currentPage === pages}
                    onClick={() => setPage(currentPage + 1)}
                  >
                    <Icon name=\"chevron\" size={16} />
                  </button>
                </div>
              </div>
            </div>
          </section>
          <div className=\"page-footer\">
            <span>
              <Icon name=\"check\" size={14} />
              Данные вашей команды под надёжной защитой
            </span>
            <span>
              Создавайте больше. Вместе с <strong>forma.</strong>
            </span>
          </div>
        </main>
      </div>
      {toast && (
        <div className=\"toast\" role=\"status\">
          <span>
            <Icon name=\"check\" size={17} />
          </span>
          {toast}
          <button aria-label=\"Закрыть уведомление\" onClick={() => setToast(\"\")}>
            <Icon name=\"close\" size={16} />
          </button>
        </div>
      )}
      {addOpen && (
        <div className=\"modal-overlay\" onClick={() => setAddOpen(false)}>
          <section
            className=\"modal\"
            role=\"dialog\"
            aria-modal=\"true\"
            aria-labelledby=\"add-title\"
            onClick={(e) => e.stopPropagation()}
          >
            <button
              className=\"modal-close\"
              aria-label=\"Закрыть\"
              onClick={() => setAddOpen(false)}
            >
              <Icon name=\"close\" />
            </button>
            <div className=\"modal-icon\">
              <Icon name=\"users\" size={26} />
            </div>
            <h2 id=\"add-title\">Новый сотрудник</h2>
            <p>Добавьте нового человека в вашу команду.</p>
            <form
              onSubmit={(e) => {
                e.preventDefault();
                if (!newName.trim()) return;
                setPeople((prev) => [
                  ...prev,
                  {
                    id: Math.max(...prev.map((p) => p.id)) + 1,
                    name: newName.trim(),
                    email: newEmail,
                    role: \"Новый сотрудник\",
                    department: newDept,
                    status: \"Активен\",
                    date: new Date().toISOString().slice(0, 10),
                    color: people.length % 8,
                  },
                ]);
                setAddOpen(false);
                setNewName(\"\");
                setNewEmail(\"\");
                setQuery(\"\");
                setStatus(\"all\");
                setDepartment(\"all\");
                setTab(\"Все сотрудники\");
                setSort({ key: \"date\", asc: false });
                setSortActive(true);
                setPage(1);
                notify(\"Сотрудник добавлен в команду\");
              }}
            >
              <label htmlFor=\"name\">Имя и фамилия</label>
              <input
                id=\"name\"
                autoFocus
                required
                placeholder=\"Например, Анна Смирнова\"
                value={newName}
                onChange={(e) => setNewName(e.target.value)}
              />
              <label htmlFor=\"email\">Рабочая почта</label>
              <input
                id=\"email\"
                type=\"email\"
                required
                placeholder=\"name@acme.design\"
                value={newEmail}
                onChange={(e) => setNewEmail(e.target.value)}
              />
              <label>Отдел</label>
              <Select
                size=\"l\"
                width=\"max\"
                value={[newDept]}
                onUpdate={(v) => setNewDept(v[0])}
                options={[
                  \"Дизайн\",
                  \"Разработка\",
                  \"Маркетинг\",
                  \"HR\",
                  \"Продажи\",
                ].map((v) => ({ value: v, content: v }))}
              />
              <div className=\"modal-actions\">
                <Button
                  size=\"l\"
                  view=\"outlined\"
                  onClick={() => setAddOpen(false)}
                >
                  Отмена
                </Button>
                <Button size=\"l\" view=\"action\" type=\"submit\">
                  Добавить сотрудника
                </Button>
              </div>
            </form>
          </section>
        </div>
      )}
      {detail && (
        <div className=\"modal-overlay\" onClick={() => setDetail(null)}>
          <section
            className=\"modal detail-modal\"
            role=\"dialog\"
            aria-modal=\"true\"
            aria-labelledby=\"detail-title\"
            onClick={(e) => e.stopPropagation()}
          >
            <button
              className=\"modal-close\"
              aria-label=\"Закрыть\"
              onClick={() => setDetail(null)}
            >
              <Icon name=\"close\" />
            </button>
            <span className={`avatar large-avatar avatar-${detail.color}`}>
              {initials(detail.name)}
            </span>
            <h2 id=\"detail-title\">{detail.name}</h2>
            <p>{detail.role}</p>
            <dl>
              <div>
                <dt>Рабочая почта</dt>
                <dd>
                  <a href={`mailto:${detail.email}`}>{detail.email}</a>
                </dd>
              </div>
              <div>
                <dt>Отдел</dt>
                <dd>{detail.department}</dd>
              </div>
              <div>
                <dt>Статус</dt>
                <dd>{detail.status}</dd>
              </div>
              <div>
                <dt>Дата добавления</dt>
                <dd>{dateLabel(detail.date)}</dd>
              </div>
            </dl>
            <Button
              view=\"action\"
              size=\"l\"
              width=\"max\"
              onClick={() => {
                toggle(detail.id);
                setDetail(null);
              }}
            >
              {selected.has(detail.id)
                ? \"Снять выделение\"
                : \"Выбрать сотрудника\"}
            </Button>
          </section>
        </div>
      )}
    </div>
  );
}
