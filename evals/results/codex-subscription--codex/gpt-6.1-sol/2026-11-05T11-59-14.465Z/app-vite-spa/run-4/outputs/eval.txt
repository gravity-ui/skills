function useStored<T>(key: string, initial: T) {
  const [value, setValue] = useState<T>(() => {
    try {
      const saved = localStorage.getItem(key);
      return saved ? JSON.parse(saved) : initial;
    } catch {
      return initial;
    }
  });
  useEffect(() => {
    localStorage.setItem(key, JSON.stringify(value));
  }, [key, value]);
  return [value, setValue] as const;
}

function Modal({
  title,
  subtitle,
  children,
  onClose,
}: {
  title: string;
  subtitle: string;
  children: ReactNode;
  onClose: () => void;
}) {
  const ref = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const previous = document.activeElement as HTMLElement;
    const overflow = document.body.style.overflow;
    document.body.style.overflow = \"hidden\";
    const first =
      ref.current?.querySelector<HTMLElement>(\"input, textarea, select\") ??
      ref.current?.querySelector<HTMLElement>(\"button\");
    first?.focus();
    const keydown = (e: KeyboardEvent) => {
      if (e.key === \"Escape\") onClose();
      if (e.key === \"Tab\") {
        const elements = Array.from(
          ref.current?.querySelectorAll<HTMLElement>(
            'button, input, textarea, select, [tabindex=\"0\"]',
          ) ?? [],
        );
        const first = elements[0];
        const last = elements[elements.length - 1];
        if (e.shiftKey && document.activeElement === first) {
          e.preventDefault();
          last?.focus();
        } else if (!e.shiftKey && document.activeElement === last) {
          e.preventDefault();
          first?.focus();
        }
      }
    };
    document.addEventListener(\"keydown\", keydown);
    return () => {
      document.body.style.overflow = overflow;
      document.removeEventListener(\"keydown\", keydown);
      previous?.focus();
    };
  }, []);
  return (
    <div
      className=\"modal-overlay\"
      onMouseDown={(e) => {
        if (e.target === e.currentTarget) onClose();
      }}
    >
      <div
        className=\"modal\"
        ref={ref}
        role=\"dialog\"
        aria-modal=\"true\"
        aria-labelledby=\"modal-title\"
      >
        <button
          className=\"icon-button modal-close\"
          onClick={onClose}
          aria-label=\"Закрыть\"
        >
          <Icon name=\"close\" />
        </button>
        <div className=\"modal-eyebrow\">ВАШЕ ПРОСТРАНСТВО</div>
        <h2 id=\"modal-title\">{title}</h2>
        <p className=\"modal-subtitle\">{subtitle}</p>
        {children}
      </div>
    </div>
  );
}
function ProjectForm({ onSave }: { onSave: (project: Project) => void }) {
  const [color, setColor] = useState(\"purple\");
  function submit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const data = new FormData(e.currentTarget);
    const name = String(data.get(\"name\")).trim();
    if (!name) return;
    onSave({
      id: crypto.randomUUID(),
      name,
      description: String(data.get(\"description\")).trim(),
      due: String(data.get(\"due\")),
      color,
      icon: color === \"green\" ? \"leaf\" : color === \"blue\" ? \"link\" : \"spark\",
      tasks: [],
    });
  }
  return (
    <form className=\"editor-form\" onSubmit={submit}>
      <label>
        Название проекта
        <input
          name=\"name\"
          required
          maxLength={70}
          placeholder=\"Например, моя большая идея\"
          autoFocus
        />
      </label>
      <label>
        Пару слов о проекте
        <textarea
          name=\"description\"
          rows={3}
          maxLength={220}
          placeholder=\"Что вы хотите создать?\"
        />
      </label>
      <label>
        Когда планируете закончить?
        <input name=\"due\" type=\"date\" required defaultValue=\"2026-11-30\" />
      </label>
      <fieldset className=\"color-picker\">
        <legend>Цвет настроения</legend>
        {[\"purple\", \"blue\", \"green\", \"orange\"].map((c) => (
          <button
            key={c}
            type=\"button\"
            aria-label={
              {
                purple: \"Фиолетовый\",
                blue: \"Синий\",
                green: \"Зелёный\",
                orange: \"Оранжевый\",
              }[c]
            }
            aria-pressed={c === color}
            className={`${c} ${c === color ? \"chosen\" : \"\"}`}
            onClick={() => setColor(c)}
          >
            {c === color && <Icon name=\"check\" size={16} />}
          </button>
        ))}
      </fieldset>
      <button className=\"primary-button form-submit\" type=\"submit\">
        <Icon name=\"plus\" size={18} />
        Создать проект
      </button>
    </form>
  );
}
function NoteForm({
  note,
  onSave,
  onDelete,
}: {
  note?: Note;
  onSave: (note: Note) => void;
  onDelete?: () => void;
}) {
  const [pinned, setPinned] = useState(note?.pinned ?? false);
  function submit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const data = new FormData(e.currentTarget);
    const title = String(data.get(\"title\")).trim();
    if (!title) return;
    onSave({
      id: note?.id ?? crypto.randomUUID(),
      title,
      content: String(data.get(\"content\")).trim(),
      tag: String(data.get(\"tag\")),
      date: new Date().toLocaleDateString(\"en-CA\"),
      pinned,
    });
  }
  return (
    <form className=\"editor-form\" onSubmit={submit}>
      <label>
        Заголовок
        <input
          name=\"title\"
          required
          maxLength={80}
          defaultValue={note?.title}
          placeholder=\"С чего начинается ваша мысль?\"
          autoFocus
        />
      </label>
      <label>
        Ваша мысль
        <textarea
          name=\"content\"
          aria-label=\"Ваша мысль\"
          rows={6}
          required
          defaultValue={note?.content}
          placeholder=\"Здесь можно просто быть собой…\"
        />
      </label>
      <div className=\"note-form-options\">
        <label>
          Категория
          <select name=\"tag\" defaultValue={note?.tag ?? \"Мысли\"}>
            {[\"Мысли\", \"Вдохновение\", \"Дизайн\", \"Личное\"].map((t) => (
              <option key={t}>{t}</option>
            ))}
          </select>
        </label>
        <button
          className={`secondary-button ${pinned ? \"pin-selected\" : \"\"}`}
          type=\"button\"
          aria-pressed={pinned}
          onClick={() => setPinned(!pinned)}
        >
          <Icon name=\"pin\" size={16} />
          {pinned ? \"Закреплена\" : \"Закрепить\"}
        </button>
      </div>
      <div className=\"modal-actions\">
        {onDelete ? (
          <button
            type=\"button\"
            className=\"text-button danger\"
            onClick={onDelete}
          >
            <Icon name=\"trash\" size={16} />
            Удалить
          </button>
        ) : (
          <span />
        )}
        <button className=\"primary-button\" type=\"submit\">
          <Icon name=\"check\" size={17} />
          Сохранить заметку
        </button>
      </div>
    </form>
  );
}
