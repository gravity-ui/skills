                      </p>
                    </span>
                  </div>
                </div>
              </>
            ) : modal === \"settings\" ? (
              <>
                <span className=\"modal-icon\">
                  <Icon name=\"settings\" size={24} />
                </span>
                <h2>Рабочее пространство</h2>
                <p>Настройте пространство для вашей команды.</p>
                <form
                  onSubmit={(e) => {
                    e.preventDefault();
                    const f = new FormData(e.currentTarget);
                    setWorkspaceName(
                      String(f.get(\"workspace\")).trim() || \"Моя команда\",
                    );
                    setModal(null);
                    setToast(\"Настройки сохранены\");
                  }}
                >
                  <label>
                    Название команды
                    <input
                      autoFocus
                      name=\"workspace\"
                      required
                      maxLength={28}
                      defaultValue={workspaceName}
                    />
                  </label>
                  <label>
                    Строк на странице по умолчанию
                    <select
                      value={pageSize}
                      onChange={(e) => setPageSize(Number(e.target.value))}
                    >
                      <option>8</option>
                      <option>16</option>
                      <option>24</option>
                    </select>
                  </label>
                  <div className=\"modal-actions\">
                    <button type=\"submit\" className=\"button primary\">
                      Сохранить
                    </button>
                  </div>
                </form>
              </>
            ) : (
              detail && (
                <>
                  <span
                    className={`avatar detail-avatar color-${detail.color}`}
                  >
                    {initials(detail.name)}
                  </span>
                  <h2>{detail.name}</h2>
                  <p>
                    {detail.role} · {detail.company}
                  </p>
                  <dl className=\"detail-list\">
                    <div>
                      <dt>Электронная почта</dt>
                      <dd>
                        <a href={`mailto:${detail.email}`}>{detail.email}</a>
                      </dd>
                    </div>
                    <div>
                      <dt>Статус</dt>
                      <dd>{detail.status}</dd>
                    </div>
                    <div>
                      <dt>Дата добавления</dt>
                      <dd>{shortDate(detail.date)}</dd>
                    </div>
                    <div>
                      <dt>Сумма сделок</dt>
                      <dd>{money(detail.amount)}</dd>
                    </div>
                  </dl>
                  <div className=\"modal-actions\">
                    <button
                      className=\"button\"
                      onClick={() =>
                        setStatus(
                          detail.id,
                          detail.status === \"В архиве\"
                            ? \"Активный\"
                            : \"В архиве\",
                        )
                      }
                    >
                      {detail.status === \"В архиве\"
                        ? \"Восстановить\"
                        : \"В архив\"}
                    </button>
                    <button
                      className=\"button primary\"
                      onClick={() => {
                        setEditing(detail);
                        setDetail(null);
                        setModal(\"add\");
                      }}
                    >
                      Редактировать
                    </button>
                  </div>
                </>
              )
            )}
          </div>
        </div>
      )}
      {toast && (
        <div className=\"toast\" role=\"status\">
          <span>
            <Icon name=\"check\" size={17} />
          </span>
          {toast}
          <button aria-label=\"Закрыть уведомление\" onClick={() => setToast(\"\")}>
            <Icon name=\"close\" size={16} />
          </button>
        </div>
      )}
    </div>
  );
}
