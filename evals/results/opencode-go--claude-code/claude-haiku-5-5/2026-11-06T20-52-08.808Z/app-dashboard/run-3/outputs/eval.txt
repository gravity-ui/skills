import { useEffect, useState } from 'react';
import { Sidebar } from './layout/Sidebar';
import { SECTIONS, isSectionId, type SectionId } from './nav';
import { HomePage, OrdersPage, PlaceholderPage } from './pages/Pages';
import { ORDERS } from './orders/mockOrders';
import './app.css';

const SIDEBAR_KEY = 'admin.sidebarCollapsed';

/** Раздел берётся из хэша (#/orders), чтобы ссылка и обновление страницы работали. */
function readSectionFromHash(): SectionId {
  const id = window.location.hash.replace(/^#\\/?/, '');
  return isSectionId(id) ? id : 'home';
}

function readCollapsed(): boolean {
  try {
    return localStorage.getItem(SIDEBAR_KEY) === 'true';
  } catch {
    return false;
  }
}

export function App() {
  const [section, setSection] = useState<SectionId>(readSectionFromHash);
  const [collapsed, setCollapsed] = useState<boolean>(readCollapsed);

  useEffect(() => {
    const onHashChange = () => setSection(readSectionFromHash());
    window.addEventListener('hashchange', onHashChange);
    return () => window.removeEventListener('hashchange', onHashChange);
  }, []);

  useEffect(() => {
    try {
      localStorage.setItem(SIDEBAR_KEY, String(collapsed));
    } catch {
      // Приватный режим или запрет хранилища: состояние просто не сохранится.
    }
  }, [collapsed]);

  const navigate = (id: SectionId) => {
    setSection(id);
    window.location.hash = `/${id}`;
  };

  const page = (() => {
    switch (section) {
      case 'home':
        return <HomePage orders={ORDERS} />;
      case 'orders':
        return <OrdersPage orders={ORDERS} />;
      case 'users':
        return <PlaceholderPage title=\"Пользователи\" />;
      case 'settings':
        return <PlaceholderPage title=\"Настройки\" />;
    }
  })();

  return (
    <div className=\"app\">
      <Sidebar
        sections={SECTIONS}
        activeId={section}
        collapsed={collapsed}
        onSelect={navigate}
        onToggle={() => setCollapsed((value) => !value)}
      />
      <main className=\"app__main\">{page}</main>
    </div>
  );
}

import { Button, Icon } from '@gravity-ui/uikit';
import { ChevronsLeft, ChevronsRight } from '@gravity-ui/icons';
import type { NavSection, SectionId } from '../nav';

interface SidebarProps {
  sections: readonly NavSection[];
  activeId: SectionId;
  collapsed: boolean;
  onSelect: (id: SectionId) => void;
  onToggle: () => void;
}

export function Sidebar({ sections, activeId, collapsed, onSelect, onToggle }: SidebarProps) {
  const toggleLabel = collapsed ? 'Развернуть меню' : 'Свернуть меню';

  return (
    <aside className={collapsed ? 'sidebar is-collapsed' : 'sidebar'}>
      <div className=\"sidebar__header\">
        {!collapsed && <span className=\"sidebar__brand\">Админка</span>}
        <Button view=\"flat\" size=\"l\" onClick={onToggle} title={toggleLabel} aria-label={toggleLabel} aria-expanded={!collapsed}>
          <Icon data={collapsed ? ChevronsRight : ChevronsLeft} size={18} />
        </Button>
      </div>

      <nav aria-label=\"Основная навигация\">
        <ul>
          {sections.map((section) => {
            const isActive = section.id === activeId;
            return (
              <li key={section.id}>
                <button
                  type=\"button\"
                  className=\"nav-item\"
                  aria-current={isActive ? 'page' : undefined}
                  aria-label={collapsed ? section.label : undefined}
                  title={collapsed ? section.label : undefined}
                  onClick={() => onSelect(section.id)}
                >
                  <Icon data={section.icon} size={18} />
                  {!collapsed && <span>{section.label}</span>}
                </button>
              </li>
            );
          })}
        </ul>
      </nav>
    </aside>
  );
}

import { STATUS_META, formatAmount, formatDate, type Order, type OrderStatus } from './model';
import { checkState, groupByStatus, moveItem, setIds, toggleId, type StatusGroup } from './tableLogic';

type ColumnId = 'number' | 'customer' | 'city' | 'items' | 'amount' | 'date' | 'status';

interface ColumnDef {
  id: ColumnId;
  title: string;
  numeric?: boolean;
  render: (order: Order) => ReactNode;
}

const COLUMNS: ColumnDef[] = [
  { id: 'number', title: 'Заказ', render: (o) => <span className=\"cell-mono\">#{o.id}</span> },
  { id: 'customer', title: 'Клиент', render: (o) => o.customer },
  { id: 'city', title: 'Город', render: (o) => o.city },
  { id: 'items', title: 'Позиции', numeric: true, render: (o) => o.itemsCount },
  { id: 'amount', title: 'Сумма', numeric: true, render: (o) => formatAmount(o.amount) },
  { id: 'date', title: 'Дата', render: (o) => formatDate(o.createdAt) },
  { id: 'status', title: 'Статус', render: (o) => <OrderStatusLabel status={o.status} /> },
];

const COLUMN_BY_ID = new Map(COLUMNS.map((column) => [column.id, column]));
const DEFAULT_COLUMN_ORDER: ColumnId[] = COLUMNS.map((column) => column.id);

export function OrderStatusLabel({ status }: { status: OrderStatus }) {
  const meta = STATUS_META[status];
  return (
    <Label theme={meta.theme} size=\"s\" icon={<Icon data={meta.icon} size={12} />}>
      {meta.label}
    </Label>
  );
}

interface OrdersTableProps {
  orders: readonly Order[];
}

export function OrdersTable({ orders }: OrdersTableProps) {
  const [columnOrder, setColumnOrder] = useState<ColumnId[]>(DEFAULT_COLUMN_ORDER);
  const [selected, setSelected] = useState<ReadonlySet<string>>(() => new Set());
  const [grouped, setGrouped] = useState(true);
  const [collapsedGroups, setCollapsedGroups] = useState<ReadonlySet<OrderStatus>>(() => new Set());
  const [dragId, setDragId] = useState<ColumnId | null>(null);
  const [dropId, setDropId] = useState<ColumnId | null>(null);

  const allIds = useMemo(() => orders.map((order) => order.id), [orders]);
  const groups = useMemo(() => groupByStatus(orders), [orders]);
  const allState = checkState(allIds, selected);
  const columnsChanged = columnOrder.some((id, index) => id !== DEFAULT_COLUMN_ORDER[index]);

  const toggleGroupCollapsed = (status: OrderStatus) => {
    setCollapsedGroups((prev) => {
      const next = new Set(prev);
      if (next.has(status)) {
        next.delete(status);
      } else {
        next.add(status);
      }
      return next;
    });
  };

  const resetDrag = () => {
    setDragId(null);
    setDropId(null);
  };

  const handleDragStart = (event: DragEvent<HTMLTableCellElement>, id: ColumnId) => {
    event.dataTransfer.effectAllowed = 'move';
    // Firefox не начинает перетаскивание без данных в dataTransfer.
    event.dataTransfer.setData('text/plain', id);
    setDragId(id);
  };

  const handleDragOver = (event: DragEvent<HTMLTableCellElement>, id: ColumnId) => {
    if (dragId === null || dragId === id) return;
    event.preventDefault();
    event.dataTransfer.dropEffect = 'move';
    setDropId(id);
  };

  const handleDrop = (event: DragEvent<HTMLTableCellElement>, id: ColumnId) => {
    event.preventDefault();
    if (dragId !== null && dragId !== id) {
      setColumnOrder((prev) => moveItem(prev, dragId, id));
    }
    resetDrag();
  };

  // Клавиатурный аналог перетаскивания: Alt + стрелки.
  const handleHeaderKeyDown = (event: KeyboardEvent<HTMLTableCellElement>, id: ColumnId) => {
    if (!event.altKey) return;
    const offset = event.key === 'ArrowLeft' ? -1 : event.key === 'ArrowRight' ? 1 : 0;
    if (offset === 0) return;
    event.preventDefault();
    setColumnOrder((prev) => {
      const index = prev.indexOf(id);
      const neighbour = prev[index + offset];
      return neighbour === undefined ? prev : moveItem(prev, id, neighbour);
    });
  };

  const dropSide = (id: ColumnId): 'before' | 'after' | null => {
    if (dragId === null || dropId !== id) return null;
    return columnOrder.indexOf(dragId) < columnOrder.indexOf(id) ? 'after' : 'before';
  };

  const renderRow = (order: Order) => {
    const isSelected = selected.has(order.id);
    return (
      <tr key={order.id} className={isSelected ? 'is-selected' : undefined} aria-selected={isSelected}>
        <td className=\"cell-check\">
          <Checkbox
            checked={isSelected}
            onUpdate={() => setSelected((prev) => toggleId(prev, order.id))}
            controlProps={{ 'aria-label': `Выбрать заказ #${order.id}` }}
          />
        </td>
        {columnOrder.map((id) => {
          const column = COLUMN_BY_ID.get(id)!;
          return (
            <td key={id} className={column.numeric ? 'is-numeric' : undefined}>
              {column.render(order)}
            </td>
          );
        })}
      </tr>
    );
  };

  const renderGroup = (group: StatusGroup) => {
    const collapsed = collapsedGroups.has(group.status);
    const ids = group.orders.map((order) => order.id);
    const state = checkState(ids, selected);
    return (
      <tbody key={group.status}>
        <tr className=\"group-row\">
          <td className=\"cell-check\">
            <Checkbox
              checked={state === 'all'}
              indeterminate={state === 'some'}
              onUpdate={() => setSelected((prev) => setIds(prev, ids, state !== 'all'))}
              controlProps={{ 'aria-label': `Выбрать все заказы: ${STATUS_META[group.status].label}` }}
            />
          </td>
          <td colSpan={columnOrder.length}>
            <button
              type=\"button\"
              className=\"group-toggle\"
              aria-expanded={!collapsed}
              onClick={() => toggleGroupCollapsed(group.status)}
            >
              <Icon data={collapsed ? ChevronRight : ChevronDown} size={16} />
              <OrderStatusLabel status={group.status} />
              <span className=\"group-count\">{group.orders.length}</span>
            </button>
          </td>
        </tr>
        {!collapsed && group.orders.map(renderRow)}
      </tbody>
    );
  };

  return (
    <section className=\"orders-table\">
      <div className=\"orders-toolbar\">
        <Switch checked={grouped} onUpdate={setGrouped} content=\"Группировать по статусу\" />
        <div className=\"orders-toolbar__right\">
          <span>Заказов: {orders.length}</span>
          {selected.size > 0 && (
            <>
              <span>Выбрано: {selected.size}</span>
              <Button view=\"flat\" size=\"s\" onClick={() => setSelected(new Set())}>
                Снять выделение
              </Button>
            </>
          )}
          {columnsChanged && (
            <Button view=\"flat\" size=\"s\" onClick={() => setColumnOrder(DEFAULT_COLUMN_ORDER)}>
              Сбросить порядок колонок
            </Button>
          )}
        </div>
      </div>

      <div className=\"orders-table__scroll\">
        <table className=\"orders-table__table\" aria-label=\"Заказы\">
          <thead>
            <tr>
              <th scope=\"col\" className=\"cell-check\">
                <Checkbox
                  checked={allState === 'all'}
                  indeterminate={allState === 'some'}
                  onUpdate={() => setSelected((prev) => setIds(prev, allIds, allState !== 'all'))}
                  controlProps={{ 'aria-label': 'Выбрать все заказы' }}
                />
              </th>
              {columnOrder.map((id) => {
                const column = COLUMN_BY_ID.get(id)!;
                const side = dropSide(id);
                const classNames = [
                  column.numeric ? 'is-numeric' : '',
                  dragId === id ? 'is-dragging' : '',
                  side ? `is-drop-${side}` : '',
                ]
                  .filter(Boolean)
                  .join(' ');
                return (
                  <th
                    key={id}
                    scope=\"col\"
                    className={classNames}
                    draggable
                    tabIndex={0}
                    title=\"Перетащите, чтобы поменять колонки местами. Или Alt + ← / →\"
                    onDragStart={(event) => handleDragStart(event, id)}
                    onDragOver={(event) => handleDragOver(event, id)}
                    onDrop={(event) => handleDrop(event, id)}
                    onDragEnd={resetDrag}
                    onKeyDown={(event) => handleHeaderKeyDown(event, id)}
                  >
                    <span className=\"th-content\">
                      <Icon data={Grip} size={12} className=\"grip\" />
                      {column.title}
                    </span>
                  </th>
                );
              })}
            </tr>
          </thead>

          {grouped ? (
            groups.map(renderGroup)
          ) : (
            <tbody>{orders.map(renderRow)}</tbody>
          )}

          {orders.length === 0 && (
            <tbody>
              <tr>
                <td colSpan={columnOrder.length + 1} className=\"orders-table__empty\">
                  Заказов нет
                </td>
              </tr>
            </tbody>
          )}
        </table>
      </div>
    </section>
  );
}


import { ORDER_STATUSES, type Order, type OrderStatus } from './model';

const FIRST_NAMES = ['Анна', 'Игорь', 'Мария', 'Дмитрий', 'Елена', 'Алексей', 'Ольга', 'Сергей', 'Наталья', 'Павел', 'Ирина', 'Кирилл'];
const LAST_NAMES = ['Соколова', 'Петров', 'Кузнецова', 'Волков', 'Морозова', 'Лебедев', 'Смирнова', 'Попов', 'Новикова', 'Фомин'];
const CITIES = ['Москва', 'Санкт-Петербург', 'Казань', 'Новосибирск', 'Екатеринбург', 'Нижний Новгород', 'Самара', 'Краснодар'];

// Доли статусов в тестовых данных; в сумме дают 1.
const STATUS_WEIGHTS: Record<OrderStatus, number> = {
  new: 0.15,
  processing: 0.25,
  shipped: 0.2,
  delivered: 0.3,
  cancelled: 0.1,
};

const DAY_MS = 24 * 60 * 60 * 1000;
const BASE_TIME = Date.UTC(2026, 10, 6);

/** Детерминированный генератор (mulberry32): одни и те же данные при каждом запуске. */
function createRandom(seed: number): () => number {
  let state = seed >>> 0;
  return () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/** Тестовые заказы. В боевом приложении их заменит запрос к API. */
export function generateOrders(count: number, seed = 20261106): Order[] {
  const random = createRandom(seed);
  const pick = <T>(items: readonly T[]): T => items[Math.floor(random() * items.length)];

  const pickStatus = (): OrderStatus => {
    let roll = random();
    for (const status of ORDER_STATUSES) {
      roll -= STATUS_WEIGHTS[status];
      if (roll < 0) return status;
    }
    return ORDER_STATUSES[ORDER_STATUSES.length - 1];
  };

  const drafts = Array.from({ length: count }, () => ({
    customer: `${pick(FIRST_NAMES)} ${pick(LAST_NAMES)}`,
    city: pick(CITIES),
    itemsCount: 1 + Math.floor(random() * 6),
    amount: Math.round((1500 + random() * 48000) / 10) * 10,
    createdAt: new Date(BASE_TIME - Math.floor(random() * 45) * DAY_MS),
    status: pickStatus(),
  }));

  // Сначала самые свежие; номер растёт вместе с датой.
  drafts.sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime());
  return drafts.map((draft, index) => ({ id: String(1000 + count - index), ...draft }));
}

import { ORDER_STATUSES, type Order, type OrderStatus } from './model';

export interface StatusGroup {
  status: OrderStatus;
  orders: Order[];
}

/** Группирует заказы по статусу. Порядок групп фиксирован, пустые группы пропускаются. */
export function groupByStatus(orders: readonly Order[]): StatusGroup[] {
  return ORDER_STATUSES.map((status) => ({
    status,
    orders: orders.filter((order) => order.status === status),
  })).filter((group) => group.orders.length > 0);
}

/**
 * Переносит `from` на место `to`. Если тащим вправо, элемент встаёт за `to`;
 * если влево, перед ним. Возвращает новый массив.
 */
export function moveItem<T>(items: readonly T[], from: T, to: T): T[] {
  const fromIndex = items.indexOf(from);
  const toIndex = items.indexOf(to);
  const next = [...items];
  if (fromIndex === -1 || toIndex === -1 || fromIndex === toIndex) return next;
  next.splice(fromIndex, 1);
  next.splice(toIndex, 0, from);
  return next;
}

export function toggleId(selected: ReadonlySet<string>, id: string): Set<string> {
  const next = new Set(selected);
  if (next.has(id)) {
    next.delete(id);
  } else {
    next.add(id);
  }
  return next;
}

export function setIds(selected: ReadonlySet<string>, ids: readonly string[], on: boolean): Set<string> {
  const next = new Set(selected);
  for (const id of ids) {
    if (on) {
      next.add(id);
    } else {
      next.delete(id);
    }
  }
  return next;
}

export type CheckState = 'none' | 'some' | 'all';

/** Состояние «общего» чекбокса: ничего, часть или всё выбрано. */
export function checkState(ids: readonly string[], selected: ReadonlySet<string>): CheckState {
  const count = ids.filter((id) => selected.has(id)).length;
  if (count === 0) return 'none';
  return count === ids.length ? 'all' : 'some';
}

    let revenue = 0;
    for (const order of orders) {
      if (order.status === 'new' || order.status === 'processing') active += 1;

    <>
      <PageHeader title=\"Главная\" />

        <Kpi label=\"Выручка без отмен\" value={formatAmount(stats.revenue)} />
      </div>
      <OrdersTable orders={orders} />
    </>
  );
}
