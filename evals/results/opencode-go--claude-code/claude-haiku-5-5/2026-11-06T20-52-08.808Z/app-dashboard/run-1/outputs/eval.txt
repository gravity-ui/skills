function SelectBox({
  checked,
  indeterminate = false,
  label,
  onChange,
}: {
  checked: boolean;
  indeterminate?: boolean;
  label: string;
  onChange: (checked: boolean) => void;
}) {
  const ref = useRef<HTMLInputElement>(null);

  useEffect(() => {
    if (ref.current) ref.current.indeterminate = indeterminate;

    <input
      ref={ref}
      type=\"checkbox\"
      className=\"select-box\"
      checked={checked}
      aria-label={label}

  );

export type DeltaTone = 'good' | 'bad' | 'neutral';

export interface KpiDelta {
  text: string;
  direction: 'up' | 'down' | 'flat';
  tone: DeltaTone;
}

export interface Kpi {
  id: string;
  label: string;
  value: string;
  /** Изменение относительно предыдущих 30 дней; undefined, если сравнивать не с чем. */
  delta?: KpiDelta;
}

const DAY_MS = 24 * 60 * 60 * 1000;
const PERIOD_DAYS = 30;

interface PeriodStats {
  count: number;
  revenue: number;
  averageCheck: number;
  cancelRate: number;
}

function periodStats(orders: Order[], from: number, to: number): PeriodStats {
  let count = 0;
  let paidCount = 0;
  let cancelled = 0;
  let revenue = 0;

  for (const order of orders) {
    const t = order.createdAt.getTime();
    if (t < from || t > to) continue;
    count += 1;
    if (order.status === 'cancelled') {
      cancelled += 1;
      continue;
    }
    paidCount += 1;
    revenue += order.total;
  }

  return {
    count,
    revenue,
    averageCheck: paidCount > 0 ? revenue / paidCount : 0,
    cancelRate: count > 0 ? cancelled / count : 0,
  };
}

function toneFor(direction: KpiDelta['direction'], goodWhenUp: boolean): DeltaTone {
  if (direction === 'flat') return 'neutral';
  return (direction === 'up') === goodWhenUp ? 'good' : 'bad';
}

function relativeDelta(current: number, previous: number, goodWhenUp: boolean): KpiDelta | undefined {
  if (previous === 0) return undefined;
  const change = current / previous - 1;
  const direction = Math.abs(change) < 0.0005 ? 'flat' : change > 0 ? 'up' : 'down';
  return { text: formatSignedPercent(change), direction, tone: toneFor(direction, goodWhenUp) };
}

function pointsDelta(current: number, previous: number, goodWhenUp: boolean): KpiDelta {
  const change = current - previous;
  const direction = Math.abs(change) < 0.0005 ? 'flat' : change > 0 ? 'up' : 'down';
  return { text: formatSignedPoints(change), direction, tone: toneFor(direction, goodWhenUp) };
}

export function computeKpis(orders: Order[], now: Date): Kpi[] {
  const end = now.getTime();
  const middle = end - PERIOD_DAYS * DAY_MS;
  const start = end - 2 * PERIOD_DAYS * DAY_MS;

  const current = periodStats(orders, middle, end);
  const previous = periodStats(orders, start, middle);
  const hasPrevious = previous.count > 0;

  return [
    {
      id: 'revenue',
      label: 'Выручка за 30 дней',
      value: formatCompactMoney(current.revenue),
      delta: hasPrevious ? relativeDelta(current.revenue, previous.revenue, true) : undefined,
    },
    {
      id: 'orders',
      label: 'Заказов за 30 дней',
      value: formatInteger(current.count),
      delta: hasPrevious ? relativeDelta(current.count, previous.count, true) : undefined,
    },
    {
      id: 'average',
      label: 'Средний чек',
      value: formatMoney(current.averageCheck),
      delta: hasPrevious ? relativeDelta(current.averageCheck, previous.averageCheck, true) : undefined,
    },
    {
      id: 'cancels',
      label: 'Доля отмен',
      value: formatPercent(current.cancelRate),
      delta: hasPrevious ? pointsDelta(current.cancelRate, previous.cancelRate, false) : undefined,
    },
  ];
}

/** Детерминированный PRNG (mulberry32), чтобы моки не менялись между запусками. */
function mulberry32(seed: number): () => number {
  let s = seed | 0;
  return () => {
    s = (s + 0x6d2b79f5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function weightedPick<T>(rand: () => number, entries: ReadonlyArray<readonly [T, number]>): T {
  const total = entries.reduce((sum, [, weight]) => sum + weight, 0);
  let r = rand() * total;
  for (const [value, weight] of entries) {
    r -= weight;
    if (r < 0) return value;
  }
  return entries[entries.length - 1][0];
}

/** Свежие заказы чаще «в работе», старые — почти всегда доставлены. */
function statusForAge(ageDays: number, rand: () => number): OrderStatus {
  if (ageDays < 3) {
    return weightedPick<OrderStatus>(rand, [
      ['new', 0.4],
      ['processing', 0.3],
      ['shipped', 0.2],
      ['cancelled', 0.1],
    ]);
  }
  if (ageDays < 10) {
    return weightedPick<OrderStatus>(rand, [
      ['new', 0.05],
      ['processing', 0.1],
      ['shipped', 0.15],
      ['delivered', 0.6],
      ['cancelled', 0.1],
    ]);
  }
  return weightedPick<OrderStatus>(rand, [
    ['shipped', 0.03],
    ['delivered', 0.87],
    ['cancelled', 0.1],
  ]);
}

/** Генерирует историю заказов за последние 60 дней, отсортированных от новых к старым. */
export function generateOrders(count = 240, seed = 20261106): Order[] {
  const rand = mulberry32(seed);
  const pick = <T>(arr: readonly T[]): T => arr[Math.floor(rand() * arr.length)];

  const drafts = Array.from({ length: count }, () => {
    const ageDays = rand() * HISTORY_DAYS;
    const items = 1 + Math.floor(rand() * 5);
    const pricePerItem = 290 + rand() * 7700;
    return {
      customer: pick(CUSTOMERS),
      createdAt: new Date(REFERENCE_NOW.getTime() - ageDays * DAY_MS),
      status: statusForAge(ageDays, rand),
      items,
      payment: pick<PaymentMethod>(['card', 'sbp', 'cod', 'invoice']),
      total: Math.round((items * pricePerItem) / 10) * 10,
    };
  });

  drafts.sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime());

  return drafts.map((draft, index) => ({
    id: String(10000 + (count - index)),
    ...draft,
  }));
}
