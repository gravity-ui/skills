const DAY_MS = 24 * 60 * 60 * 1000;

/** Детерминированный генератор псевдослучайных чисел (mulberry32), чтобы моки не менялись между перезагрузками. */
function mulberry32(seed: number): () => number {
  let state = seed >>> 0;
  return () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function createMockOrders(count = 120, seed = 20261106): Order[] {
  const rnd = mulberry32(seed);
  const pick = <T>(list: readonly T[]): T => list[Math.floor(rnd() * list.length)];
  const totalWeight = STATUS_WEIGHTS.reduce((sum, [, weight]) => sum + weight, 0);
  const pickStatus = (): OrderStatus => {
    let roll = rnd() * totalWeight;
    for (const [status, weight] of STATUS_WEIGHTS) {
      roll -= weight;
      if (roll < 0) return status;
    }
    return 'new';
  };

  const today = Date.UTC(2026, 10, 6, 12);
  const generated = Array.from({ length: count }, () => ({
    customer: pick(CUSTOMERS),
    city: pick(CITIES),
    items: 1 + Math.floor(rnd() * 6),
    amount: Math.round((1500 + rnd() * 48000) / 10) * 10,
    status: pickStatus(),
    createdAt: new Date(today - Math.floor(rnd() * 60 * DAY_MS)),
  }));

  // Сначала сортируем по дате, и только потом нумеруем: номер растёт вместе с датой.
  generated.sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime());
  return generated.map((order, index) => ({
    id: `#${1000 + count - index}`,
    ...order,
  }));
}

/** «1 заказ», «2 заказа», «5 заказов». */
export function pluralizeOrders(count: number): string {
  const mod10 = count % 10;
  const mod100 = count % 100;
  let word = 'заказов';
  if (mod10 === 1 && mod100 !== 11) word = 'заказ';
  else if (mod10 >= 2 && mod10 <= 4 && (mod100 < 12 || mod100 > 14)) word = 'заказа';
  return `${count} ${word}`;
}

export type ColumnId = 'id' | 'customer' | 'city' | 'items' | 'amount' | 'status' | 'createdAt';

  'id',

export interface StatusGroup {
  status: OrderStatus;
  orders: Order[];
}

/** Группирует заказы по статусу в заданном порядке. Пустые группы не возвращаются. */
export function groupByStatus(
  orders: readonly Order[],
  statuses: readonly OrderStatus[],
): StatusGroup[] {
  return statuses
    .map((status) => ({ status, orders: orders.filter((o) => o.status === status) }))
    .filter((group) => group.orders.length > 0);
}

/** Переносит `item` на место `target`. Если `item` стоит левее, он уходит правее `target`, и наоборот. */
export function moveItem<T>(items: readonly T[], item: T, target: T): T[] {
  const from = items.indexOf(item);
  const to = items.indexOf(target);
  if (from === -1 || to === -1 || from === to) return [...items];
  const next = [...items];
  next.splice(from, 1);
  next.splice(to, 0, item);
  return next;
}

/** Сдвигает `item` на одну позицию влево (delta = -1) или вправо (delta = 1). У края ничего не делает. */
export function shiftItem<T>(items: readonly T[], item: T, delta: -1 | 1): T[] {
  const from = items.indexOf(item);
  const neighbour = from + delta;
  if (from === -1 || neighbour < 0 || neighbour >= items.length) return [...items];
  return moveItem(items, item, items[neighbour]);
}

export type SelectionState = 'none' | 'some' | 'all';

export function selectionState(selected: ReadonlySet<string>, ids: readonly string[]): SelectionState {
  let count = 0;
  for (const id of ids) if (selected.has(id)) count += 1;
  if (count === 0) return 'none';
  return count === ids.length ? 'all' : 'some';
}

/** Возвращает новое множество, где `ids` выделены (`on = true`) или сняты (`on = false`). */
export function setIds(selected: ReadonlySet<string>, ids: readonly string[], on: boolean): Set<string> {
  const next = new Set(selected);
  for (const id of ids) {
    if (on) next.add(id);
    else next.delete(id);
  }
  return next;
}

/**
 * Выделение диапазона видимых строк от якоря до цели (Shift-клик).
 * Если якорь не видим (например, его группу свернули), выделяется только цель.
 */
export function selectRange(
  selected: ReadonlySet<string>,
  visibleIds: readonly string[],
  anchorId: string,
  targetId: string,
  on: boolean,
): Set<string> {
  const anchor = visibleIds.indexOf(anchorId);
  const target = visibleIds.indexOf(targetId);
  if (anchor === -1 || target === -1) return setIds(selected, [targetId], on);
  const from = Math.min(anchor, target);
  const to = Math.max(anchor, target);
  return setIds(selected, visibleIds.slice(from, to + 1), on);
}

interface OrderStats {
  total: number;
  cancelled: number;
  newCount: number;
  inWork: number;
  delivered: number;
  revenue: number;
}

function computeStats(orders: readonly Order[]): OrderStats {
  const stats: OrderStats = {
    total: orders.length,
    cancelled: 0,
    newCount: 0,
    inWork: 0,
    delivered: 0,
    revenue: 0,
  };
  for (const order of orders) {
    if (order.status === 'cancelled') {
      stats.cancelled += 1;
      continue;
    }
    stats.revenue += order.amount;
    if (order.status === 'new') stats.newCount += 1;
    if (order.status === 'new' || order.status === 'processing') stats.inWork += 1;
    if (order.status === 'delivered') stats.delivered += 1;
  }
  return stats;
}

function StatTile({ label, value, note }: { label: string; value: string; note?: string }) {
  return (
    <div className=\"stat-tile\">
      <Text variant=\"body-2\" color=\"secondary\">
        {label}
      </Text>
      <Text variant=\"header-2\">{value}</Text>
      {note && (
        <Text variant=\"caption-2\" color=\"secondary\">
          {note}
        </Text>
      )}
    </div>
  );
}

export function HomePage({ orders }: HomePageProps) {
  const stats = useMemo(() => computeStats(orders), [orders]);
  const deliveredShare = stats.total === 0 ? 0 : Math.round((stats.delivered / stats.total) * 100);

  return (
    <>
      <h1 className=\"page-title\">Главная</h1>
      <section className=\"stat-row\" aria-label=\"Ключевые показатели\">
        <StatTile label=\"Заказов\" value={String(stats.total)} note={`${stats.cancelled} отменено`} />
        <StatTile
          label=\"Выручка\"
          value={`${formatCompact(stats.revenue)} ₽`}
          note=\"без отменённых\"
        />
        <StatTile label=\"В работе\" value={String(stats.inWork)} note={`${stats.newCount} новых`} />
        <StatTile
          label=\"Доставлено\"
          value={String(stats.delivered)}
          note={`${deliveredShare}% от всех заказов`}
        />
      </section>
      <OrdersTable orders={orders} />
    </>
  );
}
