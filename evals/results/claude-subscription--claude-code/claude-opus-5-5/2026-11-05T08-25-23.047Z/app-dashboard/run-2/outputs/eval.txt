import './App.css';

import { Text } from '@gravity-ui/uikit';
import { useEffect, useMemo, useState } from 'react';

import { generateOrders } from './orders/data';
import { OrdersPage } from './orders/OrdersPage';
import { SECTIONS, Sidebar, type SectionId } from './Sidebar';
import { usePersistentState } from './usePersistentState';

const ORDERS_COUNT = 400;

function sectionFromHash(): SectionId {
    const id = window.location.hash.slice(1);
    return SECTIONS.some((s) => s.id === id) ? (id as SectionId) : 'orders';
}

export function App() {
    const [section, setSection] = useState<SectionId>(sectionFromHash);
    const [collapsed, setCollapsed] = usePersistentState(
        'sidebar.collapsed',
        () => window.innerWidth < 900,
        (raw) => raw === true,
    );
    const [orders, setOrders] = useState(() => generateOrders(ORDERS_COUNT));

    // Раздел живёт в URL: работают перезагрузка и кнопки «назад/вперёд».
    useEffect(() => {
        const onHashChange = () => setSection(sectionFromHash());
        window.addEventListener('hashchange', onHashChange);
        return () => window.removeEventListener('hashchange', onHashChange);
    }, []);

    const navigate = (id: SectionId) => {
        if (id !== section) window.location.hash = id;
    };

    const newOrders = useMemo(() => orders.filter((o) => o.status === 'new').length, [orders]);
    const title = SECTIONS.find((s) => s.id === section)!.title;

    useEffect(() => {
        document.title = `${title} — Админка`;
    }, [title]);

    return (
        <div className=\"layout\">
            <Sidebar
                active={section}
                onNavigate={navigate}
                collapsed={collapsed}
                onToggleCollapsed={() => setCollapsed((c) => !c)}
                badges={{ orders: newOrders }}
            />
            <main className=\"layout__main\">
                {section === 'orders' ? (
                    <OrdersPage orders={orders} onOrdersChange={setOrders} />
                ) : (
                    <section className=\"page page_stub\">
                        <Text as=\"h1\" variant=\"header-1\">
                            {title}
                        </Text>
                        <Text color=\"secondary\">Раздел в разработке.</Text>
                    </section>
                )}
            </main>
        </div>
    );
}


import { ChevronsLeft, ChevronsRight, Gear, House, Persons, ShoppingCart } from '@gravity-ui/icons';

export type SectionId = 'home' | 'users' | 'orders' | 'settings';

export const SECTIONS: { id: SectionId; title: string; icon: IconData }[] = [
    { id: 'home', title: 'Главная', icon: House },
    { id: 'users', title: 'Пользователи', icon: Persons },

interface SidebarProps {
    active: SectionId;
    onNavigate: (id: SectionId) => void;
    collapsed: boolean;
    onToggleCollapsed: () => void;
    /** Счётчики рядом с пунктами меню (например, новые заказы). */
    badges?: Partial<Record<SectionId, number>>;
}

export function Sidebar({ active, onNavigate, collapsed, onToggleCollapsed, badges = {} }: SidebarProps) {
    const toggleTitle = collapsed ? 'Развернуть меню' : 'Свернуть меню';

    return (
        <aside className={`sidebar${collapsed ? ' sidebar_collapsed' : ''}`}>
            <div className=\"sidebar__brand\">
                <span className=\"sidebar__logo\" aria-hidden>
                    A
                </span>
                <span className=\"sidebar__label sidebar__brand-name\">Админка</span>
            </div>

            <nav className=\"sidebar__nav\" aria-label=\"Основная навигация\">
                <ul className=\"sidebar__list\">
                    {SECTIONS.map((section) => {
                        const isActive = section.id === active;
                        const badge = badges[section.id];
                        return (
                            <li key={section.id}>
                                {/* В свёрнутом виде подпись скрыта — показываем её тултипом. */}
                                <Tooltip content={section.title} placement=\"right\" disabled={!collapsed} openDelay={200}>
                                    <a
                                        href={`#${section.id}`}
                                        className={`sidebar__item${isActive ? ' sidebar__item_active' : ''}`}
                                        aria-current={isActive ? 'page' : undefined}
                                        aria-label={
                                            collapsed ? (badge ? `${section.title}, ${badge} новых` : section.title) : undefined
                                        }
                                        onClick={(e) => {
                                            // Ctrl/Cmd/Shift+клик — открыть в новой вкладке/окне, как у обычной ссылки.
                                            if (e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
                                            e.preventDefault();
                                            onNavigate(section.id);
                                        }}
                                    >
                                        <span className=\"sidebar__icon\">
                                            <Icon data={section.icon} size={18} />
                                            {badge ? <span className=\"sidebar__dot\" aria-hidden /> : null}
                                        </span>
                                        <span className=\"sidebar__label\">{section.title}</span>
                                        {badge ? (
                                            <span className=\"sidebar__label sidebar__badge\">
                                                {badge}
                                                <span className=\"visually-hidden\"> новых</span>
                                            </span>
                                        ) : null}
                                    </a>
                                </Tooltip>
                            </li>
                        );
                    })}
                </ul>
            </nav>

            <Tooltip content={toggleTitle} placement=\"right\" disabled={!collapsed} openDelay={200}>
                <button
                    type=\"button\"
                    className=\"sidebar__item sidebar__toggle\"
                    onClick={onToggleCollapsed}
                    aria-expanded={!collapsed}
                    aria-label={collapsed ? toggleTitle : undefined}
                >
                    <span className=\"sidebar__icon\">
                        <Icon data={collapsed ? ChevronsRight : ChevronsLeft} size={18} />
                    </span>
                    <span className=\"sidebar__label\">{toggleTitle}</span>
                </button>
            </Tooltip>
        </aside>
    );
}

import { buildRows, sanitizeOrder } from './logic';
import { DEFAULT_COLUMN_ORDER, OrdersTable, pluralOrders, type ColumnId } from './OrdersTable';

const BULK_STATUSES: OrderStatus[] = ['processing', 'shipped', 'delivered', 'cancelled'];

const parseColumnOrder = (raw: unknown) => sanitizeOrder(raw, DEFAULT_COLUMN_ORDER);
const parseCollapsed = (raw: unknown) =>
    Array.isArray(raw) ? raw.filter((s): s is OrderStatus => STATUS_ORDER.includes(s)) : [];

interface OrdersPageProps {
    orders: Order[];
    onOrdersChange: (orders: Order[]) => void;
}

export function OrdersPage({ orders, onOrdersChange }: OrdersPageProps) {
    const [selected, setSelected] = useState<ReadonlySet<string>>(() => new Set());
    const [grouped, setGrouped] = usePersistentState('orders.grouped', true, (raw) => raw !== false);
    const [collapsedList, setCollapsedList] = usePersistentState<OrderStatus[]>('orders.collapsed', [], parseCollapsed);
    const [columnOrder, setColumnOrder] = usePersistentState<ColumnId[]>(
        'orders.columns',
        DEFAULT_COLUMN_ORDER,
        parseColumnOrder,
    );
    const [announcement, setAnnouncement] = useState('');

    const collapsed = useMemo(() => new Set(collapsedList), [collapsedList]);
    const rows = useMemo(() => buildRows(orders, grouped, collapsed), [orders, grouped, collapsed]);
    const isDefaultOrder = columnOrder.every((id, i) => id === DEFAULT_COLUMN_ORDER[i]);

    const toggleGroup = useCallback(
        (status: OrderStatus) =>
            setCollapsedList((list) => (list.includes(status) ? list.filter((s) => s !== status) : [...list, status])),
        [setCollapsedList],
    );

    const applyStatus = (status: OrderStatus) => {
        const count = selected.size;
        onOrdersChange(orders.map((o) => (selected.has(o.id) ? { ...o, status } : o)));
        setSelected(new Set());
        setAnnouncement(`Статус ${pluralOrders(count)} изменён на «${STATUS_META[status].label}»`);
    };

    const actions: ActionsPanelProps['actions'] = BULK_STATUSES.map((status) => {
        const text = status === 'cancelled' ? 'Отменить' : `→ ${STATUS_META[status].label}`;
        return {
            id: status,
            button: { props: { children: text, onClick: () => applyStatus(status) } },
            dropdown: { item: { text, action: () => applyStatus(status) } },
        };
    });

    return (
        <section className=\"page\">
            <header className=\"page__header\">
                <div>
                    <Text as=\"h1\" variant=\"header-1\">
                        Заказы
                    </Text>
                    <Text color=\"secondary\">
                        {pluralOrders(orders.length)} · перетащите заголовок колонки, чтобы поменять колонки местами
                    </Text>
                </div>
                <div className=\"page__toolbar\">
                    {!isDefaultOrder && (
                        <Button view=\"flat\" onClick={() => setColumnOrder(DEFAULT_COLUMN_ORDER)}>
                            <Icon data={ArrowRotateLeft} size={16} />
                            Вернуть порядок колонок
                        </Button>
                    )}
                    <Switch checked={grouped} onUpdate={setGrouped} content=\"Группировать по статусу\" />
                </div>
            </header>

            <div className=\"page__table\">
                <OrdersTable
                    orders={orders}
                    rows={rows}
                    columnOrder={columnOrder}
                    onColumnOrderChange={setColumnOrder}
                    selected={selected}
                    onSelectionChange={setSelected}
                    onToggleGroup={toggleGroup}
                    announce={setAnnouncement}
                />
            </div>

            {selected.size > 0 && (
                <ActionsPanel
                    className=\"page__actions\"
                    actions={actions}
                    onClose={() => setSelected(new Set())}
                    renderNote={() => `Выбрано: ${pluralOrders(selected.size)}`}
                />
            )}

            <div className=\"visually-hidden\" role=\"status\" aria-live=\"polite\">
                {announcement}
            </div>
        </section>
    );
}

interface OrdersTableProps {
    orders: readonly Order[];
    rows: readonly TableRow[];
    columnOrder: readonly ColumnId[];
    onColumnOrderChange: (order: ColumnId[]) => void;
    selected: ReadonlySet<string>;
    onSelectionChange: (next: ReadonlySet<string>) => void;
    onToggleGroup: (status: OrderStatus) => void;
    /** Сообщение для скринридеров (перенос колонки и т.п.). */
    announce: (message: string) => void;
}

type DropTarget = { id: ColumnId; side: 'before' | 'after' };

export function OrdersTable({
    orders,
    rows,
    columnOrder,
    onColumnOrderChange,
    selected,
    onSelectionChange,
    onToggleGroup,
    announce,
}: OrdersTableProps) {
    const columns = useMemo(() => columnOrder.map((id) => COLUMNS[id]), [columnOrder]);
    const allIds = useMemo(() => orders.map((o) => o.id), [orders]);
    const visibleIds = useMemo(
        () => rows.flatMap((r) => (r.kind === 'order' ? [r.order.id] : [])),
        [rows],
    );

    // --- выделение строк -------------------------------------------------
    // Колбэки стабильны (иначе memo-строки перерисуются все разом), а свежие
    // значения берём из ref.
    const latest = useRef({ selected, visibleIds, onSelectionChange });
    latest.current = { selected, visibleIds, onSelectionChange };
    const anchorRef = useRef<string | null>(null);

    const toggleRow = useCallback((id: string, withShift: boolean) => {
        const { selected, visibleIds, onSelectionChange } = latest.current;
        // Shift+клик применяет новое состояние кликнутой строки ко всему диапазону.
        const value = !selected.has(id);
        const ids = withShift ? rangeIds(visibleIds, anchorRef.current, id) : [id];
        anchorRef.current = id;
        onSelectionChange(setMany(selected, ids, value));
    }, []);

    const setGroupSelected = useCallback((ids: readonly string[], value: boolean) => {
        const { selected, onSelectionChange } = latest.current;
        onSelectionChange(setMany(selected, ids, value));
    }, []);

    const allState = checkState(allIds, selected);

    // --- перетаскивание колонок -------------------------------------------
    const [dragId, setDragId] = useState<ColumnId | null>(null);
    const [dropTarget, setDropTarget] = useState<DropTarget | null>(null);
    const gripRefs = useRef(new Map<ColumnId, HTMLButtonElement>());
    const focusAfterMove = useRef<ColumnId | null>(null);

    // Перенос DOM-узла сбрасывает фокус — возвращаем его на ручку колонки.
    useEffect(() => {
        if (focusAfterMove.current) {
            gripRefs.current.get(focusAfterMove.current)?.focus();
            focusAfterMove.current = null;
        }
    }, [columnOrder]);

    const insertIndex = (target: DropTarget) => columnOrder.indexOf(target.id) + (target.side === 'after' ? 1 : 0);
    const isNoopDrop = (from: number, target: DropTarget) => {
        const to = insertIndex(target);
        return to === from || to === from + 1;
    };

    const resetDrag = () => {
        setDragId(null);
        setDropTarget(null);
    };

    const onHeadDragStart = (e: DragEvent<HTMLTableCellElement>, id: ColumnId) => {
        e.dataTransfer.effectAllowed = 'move';
        e.dataTransfer.setData('text/plain', COLUMNS[id].title); // без этого Firefox не начнёт drag
        setDragId(id);
    };

    const onHeadDragOver = (e: DragEvent<HTMLTableCellElement>, id: ColumnId) => {
        if (!dragId) return; // тащат что-то постороннее — не принимаем
        e.preventDefault();
        e.dataTransfer.dropEffect = 'move';
        const rect = e.currentTarget.getBoundingClientRect();
        const side = e.clientX < rect.left + rect.width / 2 ? 'before' : 'after';
        if (dropTarget?.id !== id || dropTarget.side !== side) setDropTarget({ id, side });
    };

    const onHeadDrop = (e: DragEvent<HTMLTableCellElement>) => {
        e.preventDefault();
        if (dragId && dropTarget) {
            const from = columnOrder.indexOf(dragId);
            if (!isNoopDrop(from, dropTarget)) {
                const next = moveItem(columnOrder, from, insertIndex(dropTarget));
                onColumnOrderChange(next);
                announce(`Колонка «${COLUMNS[dragId].title}» перемещена на позицию ${next.indexOf(dragId) + 1} из ${next.length}`);
            }
        }
        resetDrag();
    };

    const onGripKeyDown = (e: KeyboardEvent<HTMLButtonElement>, id: ColumnId) => {
        const delta = e.key === 'ArrowLeft' ? -1 : e.key === 'ArrowRight' ? 1 : 0;
        if (!delta) return;
        e.preventDefault();
        const from = columnOrder.indexOf(id);
        const to = from + delta;
        if (to < 0 || to >= columnOrder.length) return;
        const next = moveItem(columnOrder, from, delta > 0 ? to + 1 : to);
        focusAfterMove.current = id;
        onColumnOrderChange(next);
        announce(`Колонка «${COLUMNS[id].title}» на позиции ${to + 1} из ${next.length}`);
    };

    const dragFrom = dragId ? columnOrder.indexOf(dragId) : -1;
    const visibleDropTarget = dragId && dropTarget && !isNoopDrop(dragFrom, dropTarget) ? dropTarget : null;

    return (
        <table className=\"orders\" aria-label=\"Заказы\">
            <thead>
                <tr>
                    <th className=\"orders__select-cell\" scope=\"col\">
                        <Checkbox
                            checked={allState === 'all'}
                            indeterminate={allState === 'some'}
                            onUpdate={() => setGroupSelected(allIds, allState !== 'all')}
                            controlProps={{ 'aria-label': 'Выбрать все заказы' }}
                        />
                    </th>
                    {columns.map((col) => {
                        const classes = ['orders__head'];
                        if (col.align === 'end') classes.push('orders__cell_end');
                        if (col.id === dragId) classes.push('orders__head_dragging');
                        if (visibleDropTarget?.id === col.id) classes.push(`orders__head_drop-${visibleDropTarget.side}`);
                        return (
                            <th
                                key={col.id}
                                scope=\"col\"
                                className={classes.join(' ')}
                                style={{ minWidth: col.minWidth }}
                                draggable
                                onDragStart={(e) => onHeadDragStart(e, col.id)}
                                onDragOver={(e) => onHeadDragOver(e, col.id)}
                                onDragLeave={(e) => {
                                    // Курсор ушёл с ячейки (а не на её потомка) — прячем индикатор.
                                    if (!e.currentTarget.contains(e.relatedTarget as Node | null)) setDropTarget(null);
                                }}
                                onDrop={onHeadDrop}
                                onDragEnd={resetDrag}
                            >
                                <span className=\"orders__head-inner\">
                                    <button
                                        type=\"button\"
                                        className=\"orders__grip\"
                                        ref={(el) => {
                                            if (el) gripRefs.current.set(col.id, el);
                                            else gripRefs.current.delete(col.id);
                                        }}
                                        aria-label={`Переместить колонку «${col.title}»`}
                                        aria-description=\"Стрелки влево и вправо меняют позицию колонки\"
                                        onKeyDown={(e) => onGripKeyDown(e, col.id)}
                                    >
                                        <Icon data={Grip} size={12} />
                                    </button>
                                    {col.title}
                                </span>
                            </th>
                        );
                    })}
                </tr>
            </thead>
            <tbody>
                {rows.length === 0 ? (
                    <tr>
                        <td className=\"orders__empty\" colSpan={columns.length + 1}>
                            Заказов нет
                        </td>
                    </tr>
                ) : (
                    rows.map((row) =>
                        row.kind === 'group' ? (
                            <GroupRow
                                key={`group-${row.status}`}
                                row={row}
                                colSpan={columns.length + 1}
                                checked={checkState(row.orders.map((o) => o.id), selected)}
                                onToggleCollapsed={onToggleGroup}
                                onSelect={setGroupSelected}
                            />
                        ) : (
                            <OrderRow
                                key={row.order.id}
                                order={row.order}
                                columns={columns}
                                selected={selected.has(row.order.id)}
                                onToggle={toggleRow}
                            />
                        ),
                    )
                )}
            </tbody>
        </table>
    );
}

// ---------------------------------------------------------------- строки

interface OrderRowProps {
    order: Order;
    columns: Column[];
    selected: boolean;
    onToggle: (id: string, withShift: boolean) => void;
}

const OrderRow = memo(function OrderRow({ order, columns, selected, onToggle }: OrderRowProps) {
    const onRowClick = (e: MouseEvent<HTMLTableRowElement>) => {
        // Пользователь выделял текст мышью — это не клик по строке.
        if (window.getSelection()?.toString()) return;
        onToggle(order.id, e.shiftKey);
    };

    const onSelectCellClick = (e: MouseEvent<HTMLTableCellElement>) => {
        e.stopPropagation();
        const target = e.target as HTMLElement;
        // Клик по <label> чекбокса браузер повторит кликом по <input> —
        // обрабатываем только его, чтобы не переключить строку дважды.
        if (target instanceof HTMLInputElement || !target.closest('label')) onToggle(order.id, e.shiftKey);
    };

    return (
        <tr
            className={`orders__row${selected ? ' orders__row_selected' : ''}`}
            onClick={onRowClick}
            // Shift+клик не должен выделять текст между строками.
            onMouseDown={(e) => e.shiftKey && e.preventDefault()}
        >
            <td className=\"orders__select-cell\" onClick={onSelectCellClick}>
                <Checkbox checked={selected} controlProps={{ 'aria-label': `Выбрать заказ #${order.number}` }} />
            </td>
            {columns.map((col) => (
                <td key={col.id} className={col.align === 'end' ? 'orders__cell_end' : undefined}>
                    {col.render(order)}
                </td>
            ))}
        </tr>
    );
});

interface GroupRowProps {
    row: Extract<TableRow, { kind: 'group' }>;
    colSpan: number;
    checked: CheckState;
    onToggleCollapsed: (status: OrderStatus) => void;
    onSelect: (ids: readonly string[], value: boolean) => void;
}

function GroupRow({ row, colSpan, checked, onToggleCollapsed, onSelect }: GroupRowProps) {
    const meta = STATUS_META[row.status];
    const count = row.orders.length;
    return (
        <tr className=\"orders__group\" onClick={() => onToggleCollapsed(row.status)}>
            <td colSpan={colSpan}>
                <div className=\"orders__group-inner\">
                    <span onClick={(e) => e.stopPropagation()} className=\"orders__group-check\">
                        <Checkbox
                            checked={checked === 'all'}
                            indeterminate={checked === 'some'}
                            onUpdate={() =>
                                onSelect(
                                    row.orders.map((o) => o.id),
                                    checked !== 'all',
                                )
                            }
                            controlProps={{ 'aria-label': `Выбрать все заказы со статусом «${meta.label}»` }}
                        />
                    </span>
                    <button
                        type=\"button\"
                        className=\"orders__group-toggle\"
                        aria-expanded={!row.collapsed}
                        onClick={(e) => {
                            e.stopPropagation();
                            onToggleCollapsed(row.status);
                        }}
                    >
                        <Icon data={row.collapsed ? ChevronRight : ChevronDown} size={14} />
                        <Label theme={meta.theme} size=\"xs\">
                            {meta.label}
                        </Label>
                        <span className=\"orders__group-count\">{pluralOrders(count)}</span>
                    </button>
                    <span className=\"orders__group-total\">на {formatMoney(row.total)}</span>
                </div>
            </td>
        </tr>
    );
}

export function pluralOrders(n: number) {
    const mod10 = n % 10;
    const mod100 = n % 100;
    const word =
        mod10 === 1 && mod100 !== 11
            ? 'заказ'
            : mod10 >= 2 && mod10 <= 4 && (mod100 < 12 || mod100 > 14)
              ? 'заказа'
              : 'заказов';
    return `${n} ${word}`;
}

};

const DOMAINS = ['mail.ru', 'yandex.ru', 'gmail.com', 'company.ru'];
// Веса статусов: в реальной базе доставленных больше всего, возвратов — меньше всего.
const STATUS_WEIGHTS: [OrderStatus, number][] = [
    ['new', 14],
    ['processing', 18],
    ['shipped', 16],
    ['delivered', 38],
    ['cancelled', 9],
    ['returned', 5],
];

/** Детерминированный PRNG, чтобы данные не «прыгали» между перезагрузками. */
function mulberry32(seed: number) {
    let a = seed >>> 0;
    return () => {
        a = (a + 0x6d2b79f5) >>> 0;
        let t = a;
        t = Math.imul(t ^ (t >>> 15), t | 1);
        t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
        return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
}

export function generateOrders(count: number, seed = 42, now = new Date('2026-11-05T12:00:00Z')): Order[] {
    const rnd = mulberry32(seed);
    const pick = <T,>(list: readonly T[]) => list[Math.floor(rnd() * list.length)];
    const weightTotal = STATUS_WEIGHTS.reduce((s, [, w]) => s + w, 0);
    const pickStatus = (): OrderStatus => {
        let r = rnd() * weightTotal;
        for (const [status, w] of STATUS_WEIGHTS) {
            r -= w;
            if (r < 0) return status;
        }
        return 'delivered';
    };

    const orders: Order[] = [];
    let time = now.getTime();
    for (let i = 0; i < count; i++) {
        // Заказы идут от новых к старым с интервалом от 5 минут до ~6 часов.
        time -= Math.floor(5 * 60_000 + rnd() * 6 * 3_600_000);
        const first = pick(FIRST_NAMES);
        const lastBase = pick(LAST_NAMES);
        const last = FEMALE.has(first) ? `${lastBase}а` : lastBase;
        const items = 1 + Math.floor(rnd() * rnd() * 12);
        const number = 48_000 + count - i;
        orders.push({
            id: `ord-${number}`,
            number,
            customer: `${first} ${last}`,
            email: `${TRANSLIT[first]}.${TRANSLIT[lastBase]}${Math.floor(rnd() * 90 + 10)}@${pick(DOMAINS)}`,
            createdAt: new Date(time).toISOString(),
            status: pickStatus(),
            items,
            total: Math.round((items * (400 + rnd() * 6_000)) / 10) * 10,
            payment: pick(PAYMENTS),
            city: pick(CITIES),
            manager: pick(MANAGERS),
        });
    }
    return orders;
}

describe('moveItem', () => {
    const list = ['a', 'b', 'c', 'd'];

    it('moves forward: insert before an element to the right', () => {
        expect(moveItem(list, 0, 2)).toEqual(['b', 'a', 'c', 'd']);
        expect(moveItem(list, 0, 4)).toEqual(['b', 'c', 'd', 'a']);
    });

    it('moves backward: insert before an element to the left', () => {
        expect(moveItem(list, 3, 0)).toEqual(['d', 'a', 'b', 'c']);
        expect(moveItem(list, 2, 1)).toEqual(['a', 'c', 'b', 'd']);
    });

    it('is a no-op when dropped onto itself or right after itself', () => {
        expect(moveItem(list, 1, 1)).toEqual(list);
        expect(moveItem(list, 1, 2)).toEqual(list);
    });

    it('does not mutate the input and tolerates bad indexes', () => {
        const copy = list.slice();
        moveItem(list, 0, 3);
        expect(list).toEqual(copy);
        expect(moveItem(list, 9, 0)).toEqual(list);
        expect(moveItem(list, 0, 99)).toEqual(['b', 'c', 'd', 'a']);
    });
});

import { STATUS_ORDER, type Order, type OrderStatus } from './data';

/**
 * Переносит элемент с позиции `from` так, чтобы он оказался перед элементом,
 * который до переноса стоял на позиции `insertBefore` (`list.length` — в конец).
 */
export function moveItem<T>(list: readonly T[], from: number, insertBefore: number): T[] {
    if (from < 0 || from >= list.length) return list.slice();
    const target = Math.max(0, Math.min(insertBefore, list.length));
    const result = list.slice();
    const [item] = result.splice(from, 1);
    result.splice(target > from ? target - 1 : target, 0, item);
    return result;
}

/**
 * Приводит сохранённый порядок колонок к актуальному набору: неизвестные id
 * выбрасываются, новые колонки дописываются в конец в порядке по умолчанию.
 */
export function sanitizeOrder<T extends string>(saved: unknown, defaults: readonly T[]): T[] {
    if (!Array.isArray(saved)) return defaults.slice();
    const known = new Set<string>(defaults);
    const seen = new Set<string>();
    const result: T[] = [];
    for (const id of saved) {
        if (typeof id === 'string' && known.has(id) && !seen.has(id)) {
            seen.add(id);
            result.push(id as T);
        }
    }
    for (const id of defaults) if (!seen.has(id)) result.push(id);
    return result;
}

export type TableRow =
    | { kind: 'group'; status: OrderStatus; orders: Order[]; total: number; collapsed: boolean }
    | { kind: 'order'; order: Order };

/**
 * Плоский список строк для рендера. При группировке перед заказами каждого
 * статуса идёт строка-заголовок; заказы свёрнутых групп не попадают в список.
 * Пустые группы не показываются. Порядок заказов внутри группы сохраняется.
 */
export function buildRows(orders: readonly Order[], grouped: boolean, collapsed: ReadonlySet<OrderStatus>): TableRow[] {
    if (!grouped) return orders.map((order) => ({ kind: 'order', order }));

    const byStatus = new Map<OrderStatus, Order[]>();
    for (const order of orders) {
        const bucket = byStatus.get(order.status);
        if (bucket) bucket.push(order);
        else byStatus.set(order.status, [order]);
    }

    const rows: TableRow[] = [];
    for (const status of STATUS_ORDER) {
        const groupOrders = byStatus.get(status);
        if (!groupOrders) continue;
        const isCollapsed = collapsed.has(status);
        rows.push({
            kind: 'group',
            status,
            orders: groupOrders,
            total: groupOrders.reduce((sum, o) => sum + o.total, 0),
            collapsed: isCollapsed,
        });
        if (!isCollapsed) for (const order of groupOrders) rows.push({ kind: 'order', order });
    }
    return rows;
}

export type CheckState = 'all' | 'some' | 'none';

export function checkState(ids: readonly string[], selected: ReadonlySet<string>): CheckState {
    let count = 0;
    for (const id of ids) if (selected.has(id)) count++;
    if (count === 0) return 'none';
    return count === ids.length ? 'all' : 'some';
}

/** Выставляет `value` всем `ids`. Возвращает тот же Set, если ничего не изменилось. */
export function setMany(selected: ReadonlySet<string>, ids: readonly string[], value: boolean): ReadonlySet<string> {
    const next = new Set(selected);
    for (const id of ids) {
        if (value) next.add(id);
        else next.delete(id);
    }
    return next.size === selected.size && ids.every((id) => selected.has(id) === value) ? selected : next;
}

/**
 * Ids между якорем и целевой строкой включительно — для выделения диапазона
 * по Shift+клику. Если якоря нет среди видимых строк, диапазон — одна строка.
 */
export function rangeIds(visibleIds: readonly string[], anchorId: string | null, targetId: string): string[] {
    const to = visibleIds.indexOf(targetId);
    if (to === -1) return [];
    const from = anchorId === null ? -1 : visibleIds.indexOf(anchorId);
    if (from === -1) return [targetId];
    const [start, end] = from <= to ? [from, to] : [to, from];
    return visibleIds.slice(start, end + 1);
}


import { useEffect, useState } from 'react';

/**
 * useState, который переживает перезагрузку страницы. `parse` валидирует
 * сохранённое значение — в localStorage может лежать что угодно.
 */
export function usePersistentState<T>(
    key: string,
    initial: T | (() => T),
    parse: (raw: unknown) => T = (raw) => raw as T,
) {
    const [value, setValue] = useState<T>(() => {
        try {
            const raw = window.localStorage.getItem(key);
            if (raw !== null) return parse(JSON.parse(raw));
        } catch {
            // приватный режим / битый JSON — используем значение по умолчанию
        }
        return typeof initial === 'function' ? (initial as () => T)() : initial;
    });

    useEffect(() => {
        try {
            window.localStorage.setItem(key, JSON.stringify(value));
        } catch {
            // хранилище недоступно — состояние просто не сохранится
        }
    }, [key, value]);

    return [value, setValue] as const;
}
