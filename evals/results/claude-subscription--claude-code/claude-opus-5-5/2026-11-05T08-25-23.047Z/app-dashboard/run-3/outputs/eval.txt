import './App.css';

import type {SectionId} from './Sidebar';

const DEFAULT_SECTION: SectionId = 'orders';
const ORDERS = generateOrders(320);

function readSection(): SectionId {
    const hash = window.location.hash.slice(1);
    return SECTIONS.some((s) => s.id === hash) ? (hash as SectionId) : DEFAULT_SECTION;
}

/** Активный раздел живёт в URL-хеше: переживает перезагрузку и работает с кнопками «назад/вперёд». */
function useSection() {
    const [section, setSection] = React.useState(readSection);
    React.useEffect(() => {
        const onChange = () => setSection(readSection());
        window.addEventListener('hashchange', onChange);
        return () => window.removeEventListener('hashchange', onChange);
    }, []);
    return section;
}

function SectionStub({id}: {id: SectionId}) {
    const title = SECTIONS.find((s) => s.id === id)!.title;
    return (
        <section className=\"stub\">
            <Text variant=\"header-1\" as=\"h1\">
                {title}
            </Text>
            <Text color=\"secondary\">Раздел пока пуст. Таблица заказов — в разделе «Заказы».</Text>
        </section>
    );
}

export function App() {
    const section = useSection();
    const [collapsed, setCollapsed] = usePersistentState('sidebar.collapsed', (raw) => raw === true);

    return (
        <div className=\"app\">
            <Sidebar active={section} collapsed={collapsed} onToggleCollapsed={() => setCollapsed((c) => !c)} />
            <main className=\"app__main\">
                {section === 'orders' ? <OrdersTable orders={ORDERS} /> : <SectionStub id={section} />}
            </main>
        </div>
    );
}

function plural(n: number, [one, few, many]: [string, string, string]) {
    const mod10 = n % 10;
    const mod100 = n % 100;
    if (mod10 === 1 && mod100 !== 11) return one;
    if (mod10 >= 2 && mod10 <= 4 && (mod100 < 12 || mod100 > 14)) return few;
    return many;
}

function matches(order: Order, query: string) {
    if (!query) return true;
    return (
        String(order.number).includes(query) ||
        order.customer.toLowerCase().includes(query) ||

/** Находит колонку (и её заголовок) под курсором — бросать можно на любую ячейку колонки, не только на заголовок. */
function findColumnUnderPointer(target: EventTarget | null) {
    const cell = (target as Element | null)?.closest?.('td, th') as HTMLTableCellElement | null;
    const header = cell?.closest('table')?.tHead?.rows[0]?.cells[cell.cellIndex];
    const id = header?.querySelector<HTMLElement>('[data-column-id]')?.dataset.columnId;
    return id && header ? {id, header} : null;
}

const PAYMENTS: Order['payment'][] = ['card', 'card', 'card', 'cash', 'invoice'];
const STATUS_WEIGHTS: [OrderStatus, number][] = [
    ['new', 18],
    ['processing', 22],
    ['shipped', 20],
    ['delivered', 32],
    ['cancelled', 8],
];

/** Детерминированный ГПСЧ, чтобы данные не менялись между перезагрузками. */
function mulberry32(seed: number) {
    return () => {
        seed |= 0;
        seed = (seed + 0x6d2b79f5) | 0;
        let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
        t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
        return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
}

function lastNameFor(lastName: string, female: boolean) {
    return female ? `${lastName}а` : lastName;
}

export function generateOrders(count: number, seed = 42): Order[] {
    const rnd = mulberry32(seed);
    const pick = <T,>(list: T[]) => list[Math.floor(rnd() * list.length)];
    const totalWeight = STATUS_WEIGHTS.reduce((sum, [, w]) => sum + w, 0);
    const pickStatus = () => {
        let r = rnd() * totalWeight;
        for (const [status, w] of STATUS_WEIGHTS) {
            r -= w;
            if (r < 0) return status;
        }
        return STATUS_WEIGHTS[0][0];
    };

    const now = new Date(2026, 10, 5, 18, 0).getTime();
    const orders: Order[] = [];
    for (let i = 0; i < count; i++) {
        const firstNameIndex = Math.floor(rnd() * FIRST_NAMES.length);
        const female = firstNameIndex % 2 === 0;
        const items = 1 + Math.floor(rnd() * 8);
        const number = 104_000 + count - i;
        orders.push({
            id: String(number),
            number,
            // Каждый следующий заказ — в среднем на 3 часа раньше предыдущего.
            createdAt: new Date(now - i * 3 * 3600_000 - Math.floor(rnd() * 3 * 3600_000)),
            customer: `${lastNameFor(pick(LAST_NAMES), female)} ${FIRST_NAMES[firstNameIndex]}`,
            city: pick(CITIES),
            items,
            amount: Math.round((items * (400 + rnd() * 6000)) / 10) * 10,
            payment: pick(PAYMENTS),
            status: pickStatus(),
            manager: pick(MANAGERS),
        });
    }
    return orders;
}

        day: '2-digit',
        month: '2-digit',
        year: 'numeric',
        hour: '2-digit',
        minute: '2-digit',
    }).format(date);

import type {Order, OrderStatus} from './data';
import {STATUSES} from './data';

export interface GroupRow {
    kind: 'group';
    id: string;
    status: OrderStatus;
    orderIds: string[];
    total: number;
    collapsed: boolean;
}

export interface OrderRow {
    kind: 'order';
    id: string;
    order: Order;
}

export type Row = GroupRow | OrderRow;

/**
 * Превращает список заказов в плоский список строк таблицы.
 * При группировке перед заказами каждого статуса вставляется строка-заголовок группы;
 * заказы свёрнутых групп в результат не попадают. Пустые группы не показываются.
 */
export function buildRows(orders: Order[], grouped: boolean, collapsed: ReadonlySet<OrderStatus>): Row[] {
    if (!grouped) {
        return orders.map((order) => ({kind: 'order', id: order.id, order}));
    }

    const byStatus = new Map<OrderStatus, Order[]>();
    for (const order of orders) {
        const list = byStatus.get(order.status);
        if (list) list.push(order);
        else byStatus.set(order.status, [order]);
    }

    const rows: Row[] = [];
    for (const {id: status} of STATUSES) {
        const list = byStatus.get(status);
        if (!list?.length) continue;
        const isCollapsed = collapsed.has(status);
        rows.push({
            kind: 'group',
            id: `group:${status}`,
            status,
            orderIds: list.map((o) => o.id),
            total: list.reduce((sum, o) => sum + o.amount, 0),
            collapsed: isCollapsed,
        });
        if (!isCollapsed) {
            for (const order of list) rows.push({kind: 'order', id: order.id, order});
        }
    }
    return rows;
}

export type CheckState = 'none' | 'some' | 'all';

export function getCheckState(ids: readonly string[], selected: ReadonlySet<string>): CheckState {
    if (ids.length === 0) return 'none';
    let count = 0;
    for (const id of ids) if (selected.has(id)) count++;
    if (count === 0) return 'none';
    return count === ids.length ? 'all' : 'some';
}

/** Выделяет все `ids`, если выделены не все; иначе снимает с них выделение. */
export function toggleMany(selected: ReadonlySet<string>, ids: readonly string[]): Set<string> {
    const next = new Set(selected);
    const selectAll = getCheckState(ids, selected) !== 'all';
    for (const id of ids) {
        if (selectAll) next.add(id);
        else next.delete(id);
    }
    return next;
}

/**
 * Клик по строке. Без Shift — переключает одну строку.
 * С Shift — применяет состояние якорной строки ко всему диапазону между якорем и `id`
 * в порядке `visibleIds` (как в файловых менеджерах).
 */
export function toggleWithRange(
    selected: ReadonlySet<string>,
    id: string,
    visibleIds: readonly string[],
    anchorId: string | null,
    shift: boolean,
): Set<string> {
    const next = new Set(selected);
    const from = anchorId === null ? -1 : visibleIds.indexOf(anchorId);
    const to = visibleIds.indexOf(id);

    if (!shift || from === -1 || to === -1 || from === to) {
        if (next.has(id)) next.delete(id);
        else next.add(id);
        return next;
    }

    const select = selected.has(anchorId!);
    const [start, end] = from < to ? [from, to] : [to, from];
    for (let i = start; i <= end; i++) {
        if (select) next.add(visibleIds[i]);
        else next.delete(visibleIds[i]);
    }
    return next;
}

export interface ColumnSetting {
    id: string;
    isSelected?: boolean;
}

/**
 * Переносит колонку `sourceId` до или после `targetId`.
 * Работает с полным списком настроек, так что скрытые колонки сохраняют свои места.
 */
export function moveColumn<T extends ColumnSetting>(
    settings: readonly T[],
    sourceId: string,
    targetId: string,
    side: 'before' | 'after',
): T[] {
    if (sourceId === targetId) return [...settings];
    const source = settings.find((s) => s.id === sourceId);
    if (!source || !settings.some((s) => s.id === targetId)) return [...settings];

    const rest = settings.filter((s) => s.id !== sourceId);
    const targetIndex = rest.findIndex((s) => s.id === targetId);
    rest.splice(side === 'before' ? targetIndex : targetIndex + 1, 0, source);
    return rest;
}

/** Сдвигает колонку на одну позицию среди видимых (для управления с клавиатуры). */
export function shiftColumn<T extends ColumnSetting>(settings: readonly T[], id: string, delta: -1 | 1): T[] {
    const visible = settings.filter((s) => s.isSelected !== false);
    const index = visible.findIndex((s) => s.id === id);
    const neighbour = visible[index + delta];
    if (index === -1 || !neighbour) return [...settings];
    return moveColumn(settings, id, neighbour.id, delta < 0 ? 'before' : 'after');
}

/** Приводит сохранённые настройки к актуальному набору колонок (новые — в конец, удалённые — выкидываются). */
export function normalizeSettings(saved: unknown, columnIds: readonly string[]): ColumnSetting[] {
    const known = new Set(columnIds);
    const result: ColumnSetting[] = [];
    const seen = new Set<string>();
    if (Array.isArray(saved)) {
        for (const item of saved) {
            if (item && typeof item.id === 'string' && known.has(item.id) && !seen.has(item.id)) {
                seen.add(item.id);
                result.push({id: item.id, isSelected: item.isSelected !== false});
            }
        }
    }
    for (const id of columnIds) if (!seen.has(id)) result.push({id, isSelected: true});
    return result;
}


import * as React from 'react';

/** useState с сохранением в localStorage. `parse` получает распарсенный JSON (или undefined) и обязан вернуть валидное значение. */
export function usePersistentState<T>(key: string, parse: (raw: unknown) => T) {
    const [value, setValue] = React.useState<T>(() => {
        try {
            const raw = window.localStorage.getItem(key);
            return parse(raw === null ? undefined : JSON.parse(raw));
        } catch {
            return parse(undefined);
        }
    });

    React.useEffect(() => {
        try {
            window.localStorage.setItem(key, JSON.stringify(value));
        } catch {
            // Приватный режим / переполнение — просто не сохраняем.
        }
    }, [key, value]);

    return [value, setValue] as const;
}
