import { useId, useState } from 'react';
import { Card, SegmentedRadioGroup, Text, TextArea } from '@gravity-ui/uikit';

import { MarkdownPreview } from './MarkdownPreview';

export type MarkdownViewMode = 'edit' | 'split' | 'preview';

export interface MarkdownFieldProps {
  label: string;
  value: string;
  onChange: (value: string) => void;
  name?: string;
  placeholder?: string;
  /** Максимальная длина текста. При превышении счётчик подсвечивается, поле считается невалидным */
  maxLength?: number;
  /** Текст ошибки валидации, показывается под полем */
  error?: string;
  minRows?: number;
  maxRows?: number;
  defaultMode?: MarkdownViewMode;
  disabled?: boolean;
}

const MODE_OPTIONS: { value: MarkdownViewMode; content: string }[] = [
  { value: 'edit', content: 'Редактор' },
  { value: 'split', content: 'Разделить' },
  { value: 'preview', content: 'Предпросмотр' },
];

export function MarkdownField({
  label,
  value,
  onChange,
  name,
  placeholder = 'Напишите текст. Поддерживается Markdown: **жирный**, *курсив*, `код`, списки, ссылки…',
  maxLength,
  error,
  minRows = 8,
  maxRows = 20,
  defaultMode = 'split',
  disabled,
}: MarkdownFieldProps) {
  const [mode, setMode] = useState<MarkdownViewMode>(defaultMode);
  const inputId = useId();

  const tooLong = maxLength !== undefined && value.length > maxLength;
  const invalid = Boolean(error) || tooLong;
  const showEditor = mode !== 'preview';
  const showPreview = mode !== 'edit';

  return (
    <div className=\"md-field\">
      <div className=\"md-field__header\">
        <Text as=\"label\" variant=\"subheader-1\" {...{ htmlFor: inputId }}>
          {label}
        </Text>
        <SegmentedRadioGroup<MarkdownViewMode>
          size=\"s\"
          value={mode}
          onUpdate={setMode}
          options={MODE_OPTIONS}
          aria-label=\"Режим отображения\"
        />
      </div>

      <div className={`md-field__body md-field__body_mode_${mode}`}>
        {showEditor && (
          <TextArea
            id={inputId}
            name={name}
            className=\"md-field__editor\"
            value={value}
            onUpdate={onChange}
            placeholder={placeholder}
            minRows={minRows}
            maxRows={maxRows}
            disabled={disabled}
            validationState={invalid ? 'invalid' : undefined}
            controlProps={{ spellCheck: true, 'aria-describedby': `${inputId}-hint` }}
          />
        )}
        {showPreview && (
          <Card view=\"outlined\" className=\"md-field__preview\" aria-live=\"polite\">
            <MarkdownPreview source={value} />
          </Card>
        )}
      </div>

      <div className={`md-field__footer${invalid ? ' md-field__footer_invalid' : ''}`} id={`${inputId}-hint`}>
        <Text variant=\"caption-2\" color={error ? 'danger' : 'secondary'}>
          {error ?? 'Форматирование применяется сразу, по мере ввода'}
        </Text>
        {maxLength !== undefined && (
          <Text variant=\"caption-2\" color=\"secondary\" className=\"md-field__counter\">
            {value.length} / {maxLength}
          </Text>
        )}
      </div>
    </div>
  );
}

import { useMemo } from 'react';
import { Text } from '@gravity-ui/uikit';

import { renderMarkdown } from './markdown';
import './MarkdownPreview.css';

interface MarkdownPreviewProps {
  source: string;
  className?: string;
}

export function MarkdownPreview({ source, className }: MarkdownPreviewProps) {
  const html = useMemo(() => renderMarkdown(source), [source]);

  if (!html) {
    return (
      <Text as=\"div\" color=\"hint\" className={className}>
        Здесь появится отформатированный текст
      </Text>
    );
  }

  // HTML уже очищен DOMPurify в renderMarkdown.
  return (
    <div
      className={['md-preview', className].filter(Boolean).join(' ')}
      dangerouslySetInnerHTML={{ __html: html }}
    />
  );
}

describe('renderMarkdown', () => {
  it('returns empty string for blank input', () => {
    expect(renderMarkdown('')).toBe('');
    expect(renderMarkdown('   \\n ')).toBe('');
  });

  it('renders inline formatting and headings', () => {
    const html = renderMarkdown('# Заголовок\\n\\n**жирный** *курсив* `код`');
    expect(html).toContain('<h1>Заголовок</h1>');
    expect(html).toContain('<strong>жирный</strong>');
    expect(html).toContain('<em>курсив</em>');
    expect(html).toContain('<code>код</code>');
  });

  it('renders lists, tables and line breaks', () => {
    const html = renderMarkdown('- один\\n- два\\n\\n| a | b |\\n|---|---|\\n| 1 | 2 |\\n\\nстрока\\nещё');
    expect(html).toContain('<li>один</li>');
    expect(html).toContain('<table>');
    expect(html).toContain('<br>');
  });

  it('opens links in a new tab with safe rel', () => {
    const html = renderMarkdown('[сайт](https://example.com)');
    expect(html).toContain('href=\"https://example.com\"');
    expect(html).toContain('target=\"_blank\"');
    expect(html).toContain('rel=\"noopener noreferrer\"');
  });

  it('strips scripts, event handlers and javascript: urls', () => {
    const html = renderMarkdown(
      '<script>alert(1)</script>\\n\\n<img src=\"x\" onerror=\"alert(1)\">\\n\\n[x](javascript:alert(1))',
    );
    expect(html).not.toContain('<script');
    expect(html).not.toContain('onerror');
    expect(html).not.toContain('javascript:');
  });
});


import DOMPurify from 'dompurify';
import { marked } from 'marked';

marked.use({ gfm: true, breaks: true });

// Внешние ссылки открываем в новой вкладке и защищаем от доступа к window.opener.
DOMPurify.addHook('afterSanitizeAttributes', (node) => {
  if (node.tagName === 'A' && node.hasAttribute('href')) {
    node.setAttribute('target', '_blank');
    node.setAttribute('rel', 'noopener noreferrer');
  }
});

/**
 * Преобразует markdown в безопасный HTML.
 * Скрипты, обработчики событий и javascript:-ссылки вырезаются.
 */
export function renderMarkdown(source: string): string {
  if (!source.trim()) {
    return '';
  }
  const html = marked.parse(source, { async: false });
  return DOMPurify.sanitize(html, {
    USE_PROFILES: { html: true },
    ADD_ATTR: ['target'],
  });
}
