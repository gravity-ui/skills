import * as React from 'react';
import { Icon, PlaceholderContainer } from '@gravity-ui/uikit';
import { Gear, Persons } from '@gravity-ui/icons';

import { ORDERS } from './data/orders';
import { OrdersPage } from './components/OrdersPage';
import { Sidebar, SECTIONS } from './components/Sidebar';
import type { SectionId } from './components/Sidebar';
import './App.css';

const STORAGE_KEY = 'admin.sidebar.collapsed';

export function App() {
  const [section, setSection] = React.useState<SectionId>('home');
  const [collapsed, setCollapsed] = React.useState<boolean>(() => {
    try {
      return window.localStorage.getItem(STORAGE_KEY) === '1';
    } catch {
      return false;
    }
  });

  const toggleSidebar = () =>
    setCollapsed((prev) => {
      const next = !prev;
      try {
        window.localStorage.setItem(STORAGE_KEY, next ? '1' : '0');
      } catch {
        /* storage unavailable — ignore */
      }
      return next;
    });

  const sectionTitle = SECTIONS.find((s) => s.id === section)?.title ?? '';

  return (
    <div className={`layout${collapsed ? ' layout_collapsed' : ''}`}>
      <Sidebar active={section} collapsed={collapsed} onSelect={setSection} onToggle={toggleSidebar} />
      <main className=\"layout__main\">
        {section === 'home' && (
          <OrdersPage orders={ORDERS} title=\"Главная\" description=\"Последние заказы магазина за три недели\" />
        )}
        {section === 'orders' && <OrdersPage orders={ORDERS} title=\"Заказы\" description=\"Все заказы\" />}
        {(section === 'users' || section === 'settings') && (
          <div className=\"layout__placeholder\">
            <PlaceholderContainer
              title={sectionTitle}
              description=\"Этот раздел пока в разработке.\"
              image={<Icon data={section === 'users' ? Persons : Gear} size={96} className=\"layout__placeholder-icon\" />}
              size=\"m\"
            />
          </div>
        )}
      </main>
    </div>
  );
}


import * as React from 'react';
import { ActionsPanel, Button, Icon, Label, Switch, Text, TextInput } from '@gravity-ui/uikit';
import type { ActionsPanelProps } from '@gravity-ui/uikit';

type ActionsPanelItem = ActionsPanelProps['actions'][number];
import { ArrowsRotateLeft, ChevronDown, ChevronRight, FileArrowDown, Magnifier, TrashBin, Xmark } from '@gravity-ui/icons';

import type { Order, OrderStatus } from '../data/orders';
import { STATUS_META } from '../data/orders';
import { OrdersTable, useColumnOrder } from './OrdersTable';
import { filterOrders, formatMoney, groupByStatus } from './tableUtils';

interface OrdersPageProps {
  orders: Order[];
  title: string;
  description?: string;
}

export function OrdersPage({ orders, title, description }: OrdersPageProps) {
  const [query, setQuery] = React.useState('');
  const [grouped, setGrouped] = React.useState(true);
  const [collapsedGroups, setCollapsedGroups] = React.useState<Set<OrderStatus>>(() => new Set());
  const [selectedIds, setSelectedIds] = React.useState<string[]>([]);
  const { columns, dragState, reset: resetColumns, isDefault: columnsDefault } = useColumnOrder();

  const visible = React.useMemo(() => filterOrders(orders, query), [orders, query]);
  const groups = React.useMemo(() => (grouped ? groupByStatus(visible) : null), [grouped, visible]);

  // Drop selection of rows that are no longer displayed (e.g. after a search).
  React.useEffect(() => {
    const visibleIds = new Set(visible.map((o) => o.id));
    setSelectedIds((prev) => (prev.every((id) => visibleIds.has(id)) ? prev : prev.filter((id) => visibleIds.has(id))));
  }, [visible]);

  const toggleGroup = (status: OrderStatus) =>
    setCollapsedGroups((prev) => {
      const next = new Set(prev);
      if (next.has(status)) next.delete(status);
      else next.add(status);
      return next;
    });

  const clearSelection = () => setSelectedIds([]);

  const selectedTotal = React.useMemo(
    () => orders.filter((o) => selectedIds.includes(o.id)).reduce((sum, o) => sum + o.amount, 0),
    [orders, selectedIds],
  );

  const actions: ActionsPanelItem[] = [
    {
      id: 'export',
      button: {
        props: { children: [<Icon key=\"i\" data={FileArrowDown} size={16} />, 'Экспорт'], onClick: () => {} },
      },
      dropdown: { item: { action: () => {}, text: 'Экспорт' } },
    },
    {
      id: 'cancel',
      button: {
        props: { children: [<Icon key=\"i\" data={Xmark} size={16} />, 'Отменить заказы'], onClick: () => {} },
      },
      dropdown: { item: { action: () => {}, text: 'Отменить заказы' } },
    },
    {
      id: 'delete',
      collapsed: true,
      button: { props: { children: [<Icon key=\"i\" data={TrashBin} size={16} />, 'Удалить'], onClick: () => {} } },
      dropdown: { item: { action: () => {}, text: 'Удалить', theme: 'danger' } },
    },
  ];

  return (
    <div className=\"orders\">
      <header className=\"orders__head\">
        <div>
          <Text as=\"h1\" variant=\"header-2\" className=\"orders__title\">
            {title}
          </Text>
          {description && (
            <Text as=\"p\" variant=\"body-1\" color=\"secondary\" className=\"orders__description\">
              {description}
            </Text>
          )}
        </div>
        <Text variant=\"body-1\" color=\"secondary\">
          Всего заказов: {orders.length}
        </Text>
      </header>

      <div className=\"orders__toolbar\" role=\"toolbar\" aria-label=\"Фильтры таблицы\">
        <TextInput
          className=\"orders__search\"
          size=\"l\"
          placeholder=\"Поиск по номеру, клиенту, городу\"
          value={query}
          onUpdate={setQuery}
          hasClear
          startContent={<Icon data={Magnifier} size={16} className=\"orders__search-icon\" />}
        />
        <div className=\"orders__toolbar-right\">
          <Switch size=\"l\" checked={grouped} onUpdate={setGrouped} content=\"Группировать по статусу\" />
          {!columnsDefault && (
            <Button view=\"outlined\" size=\"l\" onClick={resetColumns} title=\"Вернуть порядок колонок по умолчанию\">
              <Icon data={ArrowsRotateLeft} size={16} />
              Сбросить колонки
            </Button>
          )}
        </div>
      </div>

      <Text variant=\"caption-2\" color=\"hint\" className=\"orders__hint\">
        Перетаскивайте заголовки колонок, чтобы менять их порядок. Клик по строке выделяет её; Shift + клик по чекбоксу
        выделяет диапазон.
      </Text>

      {groups ? (
        <div className=\"orders__groups\">
          {groups.length === 0 && (
            <div className=\"orders__empty\">
              <Text color=\"secondary\">Заказы не найдены</Text>
            </div>
          )}
          {groups.map((group) => {
            const meta = STATUS_META[group.status];
            const isCollapsed = collapsedGroups.has(group.status);
            const selectedInGroup = group.orders.filter((o) => selectedIds.includes(o.id)).length;
            const panelId = `orders-group-${group.status}`;
            return (
              <section key={group.status} className=\"orders__group\" aria-labelledby={`${panelId}-title`}>
                <button
                  type=\"button\"
                  className=\"orders__group-head\"
                  onClick={() => toggleGroup(group.status)}
                  aria-expanded={!isCollapsed}
                  aria-controls={panelId}
                >
                  <Icon data={isCollapsed ? ChevronRight : ChevronDown} size={16} />
                  <Label theme={meta.theme} size=\"m\">
                    {meta.title}
                  </Label>
                  <Text id={`${panelId}-title`} variant=\"subheader-1\">
                    {group.orders.length} {pluralOrders(group.orders.length)}
                  </Text>
                  <Text variant=\"body-1\" color=\"secondary\">
                    на сумму {formatMoney(group.total)}
                  </Text>
                  {selectedInGroup > 0 && (
                    <Text variant=\"body-1\" color=\"brand\" className=\"orders__group-selected\">
                      выбрано {selectedInGroup}
                    </Text>
                  )}
                </button>
                {!isCollapsed && (
                  <div id={panelId} className=\"orders__group-body\">
                    <OrdersTable
                      orders={group.orders}
                      columns={columns}
                      dragState={dragState}
                      selectedIds={selectedIds}
                      onSelectionChange={setSelectedIds}
                    />
                  </div>
                )}
              </section>
            );
          })}
        </div>
      ) : (
        <div className=\"orders__single\">
          <OrdersTable
            orders={visible}
            columns={columns}
            dragState={dragState}
            selectedIds={selectedIds}
            onSelectionChange={setSelectedIds}
          />
        </div>
      )}

      {selectedIds.length > 0 && (
        <div className=\"orders__actions\">
          <ActionsPanel
            actions={actions}
            onClose={clearSelection}
            renderNote={() => (
              <span>
                Выбрано {selectedIds.length} {pluralOrders(selectedIds.length)} на {formatMoney(selectedTotal)}
              </span>
            )}
          />
        </div>
      )}
    </div>
  );
}

function pluralOrders(n: number) {
  const mod10 = n % 10;
  const mod100 = n % 100;
  if (mod10 === 1 && mod100 !== 11) return 'заказ';
  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 10 || mod100 >= 20)) return 'заказа';
  return 'заказов';
}

/* ---------- column drag & drop ---------- */

interface ColumnDragState {
  draggingId: ColumnId | null;
  overId: ColumnId | null;
  onDragStart: (id: ColumnId, e: React.DragEvent) => void;
  onDragOver: (id: ColumnId, e: React.DragEvent) => void;
  onDrop: (id: ColumnId, e: React.DragEvent) => void;
  onDragEnd: () => void;
}

const ColumnDragContext = React.createContext<ColumnDragState | null>(null);

function DraggableHeader({ id, title }: { id: ColumnId; title: string }) {
  const drag = React.useContext(ColumnDragContext);
  if (!drag) return <>{title}</>;
  return (
    <div
      className={`orders__header${drag.draggingId === id ? ' orders__header_dragging' : ''}`}
      draggable
      title=\"Перетащите, чтобы поменять колонки местами\"
      onDragStart={(e) => drag.onDragStart(id, e)}
      onDragOver={(e) => drag.onDragOver(id, e)}
      onDragEnter={(e) => drag.onDragOver(id, e)}
      onDrop={(e) => drag.onDrop(id, e)}
      onDragEnd={drag.onDragEnd}
    >
      <Icon data={Grip} size={12} className=\"orders__grip\" />
      <span>{title}</span>
    </div>
  );
}

export function useColumnOrder(initial: ColumnId[] = DEFAULT_COLUMN_ORDER) {
  const [columnOrder, setColumnOrder] = React.useState<ColumnId[]>(initial);
  const [draggingId, setDraggingId] = React.useState<ColumnId | null>(null);
  const [overId, setOverId] = React.useState<ColumnId | null>(null);

  const dragState = React.useMemo<ColumnDragState>(
    () => ({
      draggingId,
      overId,
      onDragStart: (id, e) => {
        e.dataTransfer.effectAllowed = 'move';
        e.dataTransfer.setData('text/plain', id);
        setDraggingId(id);
      },
      onDragOver: (id, e) => {
        if (!draggingId) return;
        e.preventDefault();
        e.dataTransfer.dropEffect = 'move';
        if (id !== overId) setOverId(id);
      },
      onDrop: (id, e) => {
        e.preventDefault();
        const from = (e.dataTransfer.getData('text/plain') as ColumnId) || draggingId;
        if (from) setColumnOrder((prev) => reorderColumns(prev, from, id) as ColumnId[]);
        setDraggingId(null);
        setOverId(null);
      },
      onDragEnd: () => {
        setDraggingId(null);
        setOverId(null);
      },
    }),
    [draggingId, overId],
  );

  const columns = React.useMemo<TableColumnConfig<Order>[]>(
    () =>
      columnOrder.map((id) => {
        const def = COLUMN_DEFS[id];
        const mods = [
          overId === id && draggingId !== id ? 'orders__col_over' : '',
          draggingId === id ? 'orders__col_dragging' : '',
        ]
          .filter(Boolean)
          .join(' ');
        return {
          id,
          name: () => <DraggableHeader id={id} title={def.title} />,
          template: def.template,
          width: def.width,
          align: def.align,
          className: mods || undefined,
          primary: id === 'id',
        };
      }),
    [columnOrder, draggingId, overId],
  );

  const reset = React.useCallback(() => setColumnOrder(DEFAULT_COLUMN_ORDER), []);
  const isDefault = columnOrder.join() === DEFAULT_COLUMN_ORDER.join();

  return { columns, columnOrder, dragState, reset, isDefault };
}

/** Move column `fromId` so that it lands at the position of `toId`. */
export function reorderColumns(order: string[], fromId: string, toId: string): string[] {
  if (fromId === toId) return order;
  const from = order.indexOf(fromId);
  const to = order.indexOf(toId);
  if (from === -1 || to === -1) return order;
  const next = order.slice();
  next.splice(from, 1);
  next.splice(to, 0, fromId);
  return next;
}

export interface OrderGroup {
  status: OrderStatus;
  orders: Order[];
  total: number;
}

/** Group orders by status in canonical status order; empty groups are omitted. */
export function groupByStatus(orders: Order[]): OrderGroup[] {
  const map = new Map<OrderStatus, Order[]>();
  for (const order of orders) {
    const list = map.get(order.status);
    if (list) list.push(order);
    else map.set(order.status, [order]);
  }
  return STATUS_ORDER.filter((s) => map.has(s)).map((status) => {
    const list = map.get(status)!;
    return { status, orders: list, total: list.reduce((sum, o) => sum + o.amount, 0) };
  });
}

export function filterOrders(orders: Order[], query: string): Order[] {
  const q = query.trim().toLowerCase();
  if (!q) return orders;
  return orders.filter(
    (o) =>
      o.id.toLowerCase().includes(q) ||
      o.customer.toLowerCase().includes(q) ||
      o.email.toLowerCase().includes(q) ||
      o.city.toLowerCase().includes(q),
  );
}

// Deterministic pseudo-random so the table looks the same on every reload.
function seeded(seed: number) {
  let s = seed;
  return () => {
    s = (s * 1103515245 + 12345) & 0x7fffffff;
    return s / 0x7fffffff;
  };
}

export function generateOrders(count = 48): Order[] {
  const rnd = seeded(2024);
  const base = new Date('2026-11-05T09:00:00').getTime();
  return Array.from({ length: count }, (_, i) => {
    const [customer, email] = customers[Math.floor(rnd() * customers.length)];
    const items = 1 + Math.floor(rnd() * 7);
    const amount = Math.round((900 + rnd() * 24000) * items) / 1;
    const daysAgo = Math.floor(rnd() * 21);
    const createdAt = new Date(base - daysAgo * 86_400_000 - Math.floor(rnd() * 36_000_000)).toISOString();
    return {
      id: `ORD-${String(10240 + count - i).padStart(5, '0')}`,
      customer,
      email,
      city: cities[Math.floor(rnd() * cities.length)],
      createdAt,
      items,
      amount,
      payment: payments[Math.floor(rnd() * payments.length)],
      status: statuses[Math.floor(rnd() * statuses.length)],
    };
  }).sort((a, b) => (a.createdAt < b.createdAt ? 1 : -1));
}
