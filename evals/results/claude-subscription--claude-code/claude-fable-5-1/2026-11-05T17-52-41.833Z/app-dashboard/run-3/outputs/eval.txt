function downloadCsv(orders: Order[]) {
  const blob = new Blob(['﻿' + ordersToCsv(orders)], { type: 'text/csv;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = url;
  link.download = `orders-${new Date().toISOString().slice(0, 10)}.csv`;
  link.click();
  URL.revokeObjectURL(url);
}

export function OrdersTable() {
  const [orders, setOrders] = React.useState<Order[]>(() => generateOrders());
  const [selectedIds, setSelectedIds] = React.useState<string[]>([]);
  const [columnOrder, setColumnOrder] = React.useState<ColumnId[]>(DEFAULT_COLUMN_ORDER);
  const [grouped, setGrouped] = React.useState(true);
  const [collapsedGroups, setCollapsedGroups] = React.useState<ReadonlySet<OrderStatus>>(() => new Set());

  // ---- column drag & drop -------------------------------------------------
  const [dragId, setDragId] = React.useState<string | null>(null);
  const [over, setOver] = React.useState<{ id: string; side: DropSide } | null>(null);

  const dndApi = React.useMemo<ColumnDndApi>(
    () => ({
      dragId,
      over,
      startDrag: (id) => setDragId(id),
      endDrag: () => {
        setDragId(null);
        setOver(null);
      },
      shift: (id, direction) =>
        setColumnOrder((current) => shiftColumn(current, id, direction) as ColumnId[]),
    }),
    [dragId, over],
  );

  const handleDragOver = (event: React.DragEvent<HTMLDivElement>) => {
    if (!dragId) return;
    const th = (event.target as Element).closest('th');
    const targetId = th?.querySelector<HTMLElement>('[data-column-id]')?.dataset.columnId;
    if (!th || !targetId) {
      setOver(null);
      return;
    }
    event.preventDefault();
    event.dataTransfer.dropEffect = 'move';
    const rect = th.getBoundingClientRect();
    const side: DropSide = event.clientX < rect.left + rect.width / 2 ? 'before' : 'after';
    // Don't show an indicator for drops that would not change anything.
    if (moveColumn(columnOrder, dragId, targetId, side) === columnOrder) {
      setOver(null);
      return;
    }
    setOver((prev) => (prev && prev.id === targetId && prev.side === side ? prev : { id: targetId, side }));
  };

  const handleDrop = (event: React.DragEvent<HTMLDivElement>) => {
    event.preventDefault();
    if (dragId && over) {
      setColumnOrder((current) => moveColumn(current, dragId, over.id, over.side) as ColumnId[]);
    }
    setDragId(null);
    setOver(null);
  };

  const handleDragLeave = (event: React.DragEvent<HTMLDivElement>) => {
    if (!event.currentTarget.contains(event.relatedTarget as Node | null)) {
      setOver(null);
    }
  };

  const columns = React.useMemo<TableColumnConfig<Order>[]>(
    () =>
      columnOrder.map((id) => {
        const def = COLUMN_DEFS[id];
        const align = def.config.align === 'end' || def.config.align === 'right' ? 'end' : 'start';
        return {
          ...def.config,
          id,
          meta: { ...def.config.meta, displayName: def.title },
          name: () => <ColumnHeader id={id} title={def.title} align={align} />,
        };
      }),
    [columnOrder],
  );

  const isDefaultOrder = columnOrder.every((id, index) => id === DEFAULT_COLUMN_ORDER[index]);

  // ---- selection ------------------------------------------------------------
  const selectedSet = React.useMemo(() => new Set(selectedIds), [selectedIds]);
  const toggleRow = React.useCallback((order: Order) => {
    setSelectedIds((current) =>
      current.includes(order.id) ? current.filter((id) => id !== order.id) : [...current, order.id],
    );
  }, []);
  const clearSelection = () => setSelectedIds([]);

  const selectedOrders = React.useMemo(
    () => orders.filter((order) => selectedSet.has(order.id)),
    [orders, selectedSet],
  );

  const deleteSelected = () => {
    setOrders((current) => current.filter((order) => !selectedSet.has(order.id)));
    setSelectedIds([]);
  };

  const panelActions: ActionsPanelProps['actions'] = [
    {
      id: 'export',
      button: {
        props: {
          view: 'flat-contrast',
          children: [<Icon key=\"icon\" data={ArrowDownToLine} size={16} />, 'Экспорт CSV'],
          onClick: () => downloadCsv(selectedOrders),
        },
      },
      dropdown: { item: { text: 'Экспорт CSV', action: () => downloadCsv(selectedOrders) } },
    },
    {
      id: 'delete',
      button: {
        props: {
          view: 'flat-contrast',
          children: [<Icon key=\"icon\" data={TrashBin} size={16} />, 'Удалить'],
          onClick: deleteSelected,
        },
      },
      dropdown: { item: { text: 'Удалить', theme: 'danger', action: deleteSelected } },
    },
  ];

  // ---- grouping -------------------------------------------------------------
  const groups = React.useMemo(() => groupOrdersByStatus(orders), [orders]);
  const allCollapsed = groups.length > 0 && groups.every((group) => collapsedGroups.has(group.status));

  const toggleGroup = (status: OrderStatus) => {
    setCollapsedGroups((current) => {
      const next = new Set(current);
      if (next.has(status)) next.delete(status);
      else next.add(status);
      return next;
    });
  };
  const toggleAllGroups = () => {
    setCollapsedGroups(allCollapsed ? new Set() : new Set(groups.map((group) => group.status)));
  };

  const renderTable = (data: Order[], ariaLabel: string) => (
    <SelectableTable
      data={data}
      columns={columns}
      width=\"max\"
      getRowDescriptor={getRowDescriptor}
      selectedIds={selectedIds}
      onSelectionChange={setSelectedIds}
      onRowClick={toggleRow}
      emptyMessage=\"Нет заказов\"
      aria-label={ariaLabel}
      className=\"orders-table__table\"
    />
  );

  return (
    <ColumnDndContext.Provider value={dndApi}>
      <section className=\"orders-table\" aria-labelledby=\"orders-title\">
        <header className=\"orders-table__toolbar\">
          <div className=\"orders-table__heading\">
            <Text as=\"h1\" variant=\"header-2\" id=\"orders-title\" className=\"orders-table__title\">
              Заказы
            </Text>
            <Text color=\"secondary\">
              {orders.length} {pluralize(orders.length, ORDER_FORMS)}
            </Text>
          </div>
          <div className=\"orders-table__controls\">
            <Switch size=\"m\" checked={grouped} onUpdate={setGrouped} content=\"Группировать по статусу\" />
            {grouped && groups.length > 0 && (
              <Button view=\"flat\" size=\"m\" onClick={toggleAllGroups}>
                {allCollapsed ? 'Развернуть все' : 'Свернуть все'}
              </Button>
            )}
            <Button view=\"flat\" size=\"m\" disabled={isDefaultOrder} onClick={() => setColumnOrder(DEFAULT_COLUMN_ORDER)}>
              Сбросить порядок колонок
            </Button>
          </div>
        </header>
        <Text color=\"hint\" className=\"orders-table__hint\">
          Перетащите заголовок колонки, чтобы изменить порядок. Клик по строке выделяет её, Shift + чекбокс выделяет
          диапазон.
        </Text>

        <div
          className={`orders-table__body${dragId ? ' orders-table__body_dragging' : ''}`}
          onDragOver={handleDragOver}
          onDrop={handleDrop}
          onDragLeave={handleDragLeave}
        >
          {grouped ? (
            groups.length === 0 ? (
              renderTable([], 'Заказы')
            ) : (
              groups.map((group) => {
                const meta = STATUS_META[group.status];
                const collapsed = collapsedGroups.has(group.status);
                const selectedInGroup = group.orders.filter((order) => selectedSet.has(order.id)).length;
                const panelId = `orders-group-${group.status}`;
                return (
                  <section key={group.status} className=\"orders-group\" aria-label={`Статус: ${meta.title}`}>
                    <button
                      type=\"button\"
                      className=\"orders-group__header\"
                      aria-expanded={!collapsed}
                      aria-controls={panelId}
                      onClick={() => toggleGroup(group.status)}
                    >
                      <ArrowToggle direction={collapsed ? 'right' : 'bottom'} size={16} />
                      <Label theme={meta.theme} size=\"m\">
                        {meta.title}
                      </Label>
                      <Text variant=\"body-1\" className=\"orders-group__count\">
                        {group.orders.length} {pluralize(group.orders.length, ORDER_FORMS)}
                      </Text>
                      <Text color=\"secondary\" className=\"orders-group__total\">
                        {formatMoney(group.total)}
                      </Text>
                      {selectedInGroup > 0 && (
                        <Text color=\"info\" className=\"orders-group__selected\">
                          выбрано: {selectedInGroup}
                        </Text>
                      )}
                    </button>
                    <div id={panelId} hidden={collapsed}>
                      {!collapsed && renderTable(group.orders, `Заказы со статусом «${meta.title}»`)}
                    </div>
                  </section>
                );
              })
            )
          ) : (
            renderTable(orders, 'Заказы')
          )}
        </div>

        {selectedIds.length > 0 && (
          <div className=\"orders-table__actions\">
            <ActionsPanel
              actions={panelActions}
              onClose={clearSelection}
              renderNote={() => `Выбрано: ${selectedIds.length} ${pluralize(selectedIds.length, ORDER_FORMS)}`}
            />
          </div>
        )}
      </section>
    </ColumnDndContext.Provider>
  );
}


import { Grip } from '@gravity-ui/icons';
import { Icon } from '@gravity-ui/uikit';
import * as React from 'react';

import type { DropSide } from './utils';

export interface ColumnDndApi {
  /** Column currently being dragged (HTML5 DnD). */
  dragId: string | null;
  /** Column the pointer is over and on which side the dragged column will land. */
  over: { id: string; side: DropSide } | null;
  startDrag: (id: string) => void;
  endDrag: () => void;
  /** Keyboard alternative to drag & drop. */
  shift: (id: string, direction: -1 | 1) => void;
}

export const ColumnDndContext = React.createContext<ColumnDndApi | null>(null);

interface ColumnHeaderProps {
  id: string;
  title: string;
  align?: 'start' | 'end';
}

/**
 * Draggable table header. `dragover`/`drop` are handled by the table container
 * (event delegation), so the whole `<th>` works as a drop zone, not just this element.
 */
export function ColumnHeader({ id, title, align = 'start' }: ColumnHeaderProps) {
  const dnd = React.useContext(ColumnDndContext);
  if (!dnd) {
    throw new Error('ColumnHeader must be rendered inside ColumnDndContext');
  }

  const dragging = dnd.dragId === id;
  const overSide = dnd.over?.id === id ? dnd.over.side : null;
  const className = [
    'column-header',
    `column-header_align_${align}`,
    dragging ? 'column-header_dragging' : '',
    overSide ? `column-header_over_${overSide}` : '',
  ]
    .filter(Boolean)
    .join(' ');

  const handleKeyDown = (event: React.KeyboardEvent<HTMLDivElement>) => {
    if (!event.altKey) return;
    if (event.key === 'ArrowLeft' || event.key === 'ArrowRight') {
      event.preventDefault();
      dnd.shift(id, event.key === 'ArrowLeft' ? -1 : 1);
    }
  };

  return (
    <div
      className={className}
      data-column-id={id}
      draggable
      role=\"button\"
      tabIndex={0}
      aria-label={`${title}: переместить колонку`}
      aria-roledescription=\"перетаскиваемый заголовок колонки\"
      aria-keyshortcuts=\"Alt+ArrowLeft Alt+ArrowRight\"
      title=\"Перетащите, чтобы изменить порядок колонок. Alt + ← / → — переместить с клавиатуры\"
      onDragStart={(event) => {
        event.dataTransfer.effectAllowed = 'move';
        event.dataTransfer.setData('text/plain', id);
        dnd.startDrag(id);
      }}
      onDragEnd={dnd.endDrag}
      onKeyDown={handleKeyDown}
    >
      <Icon data={Grip} size={12} className=\"column-header__grip\" />
      <span className=\"column-header__title\">{title}</span>
    </div>
  );
}

/** Weighted status distribution: new orders are the most common. */
const STATUS_WEIGHTS = [5, 4, 3, 6, 2];

/** Small seeded PRNG (mulberry32) so the demo data is stable between reloads. */
function mulberry32(seed: number) {
  return () => {
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function pickWeighted<T>(items: T[], weights: number[], rnd: number): T {
  const sum = weights.reduce((a, b) => a + b, 0);
  let acc = rnd * sum;
  for (let i = 0; i < items.length; i++) {
    acc -= weights[i];
    if (acc < 0) return items[i];
  }
  return items[items.length - 1];
}

export function generateOrders(count = 48, seed = 2026): Order[] {
  const rand = mulberry32(seed);
  const base = Date.UTC(2026, 10, 5, 12); // 5 Nov 2026, fixed so dates are deterministic
  const dayMs = 24 * 60 * 60 * 1000;
  const orders: Order[] = [];
  for (let i = 0; i < count; i++) {
    const daysAgo = Math.floor(rand() * 30);
    const minutes = Math.floor(rand() * 24 * 60);
    const itemsCount = 1 + Math.floor(rand() * 8);
    const avgPrice = 500 + Math.floor(rand() * 9000);
    orders.push({
      id: `ORD-${10240 + count - i}`,
      customer: CUSTOMERS[Math.floor(rand() * CUSTOMERS.length)],
      city: CITIES[Math.floor(rand() * CITIES.length)],
      createdAt: new Date(base - daysAgo * dayMs - minutes * 60 * 1000).toISOString(),
      itemsCount,
      total: Math.round((itemsCount * avgPrice) / 10) * 10,
      payment: PAYMENTS[Math.floor(rand() * PAYMENTS.length)],
      status: pickWeighted(STATUS_ORDER, STATUS_WEIGHTS, rand()),
    });
  }
  return orders.sort((a, b) => (a.createdAt < b.createdAt ? 1 : -1));
}

import { STATUS_ORDER, type Order, type OrderStatus } from './types';

export type DropSide = 'before' | 'after';

/**
 * Returns a new column order with `sourceId` placed before/after `targetId`.
 * Unknown ids or a no-op move return the original array (same reference).
 */
export function moveColumn(order: string[], sourceId: string, targetId: string, side: DropSide): string[] {
  if (sourceId === targetId) return order;
  const from = order.indexOf(sourceId);
  const targetIndex = order.indexOf(targetId);
  if (from === -1 || targetIndex === -1) return order;

  const next = order.filter((id) => id !== sourceId);
  let to = next.indexOf(targetId);
  if (side === 'after') to += 1;
  if (to === from) return order;
  next.splice(to, 0, sourceId);
  return next;
}

/** Moves a column one position to the left or right; stays put at the edges. */
export function shiftColumn(order: string[], id: string, direction: -1 | 1): string[] {
  const index = order.indexOf(id);
  const neighbour = order[index + direction];
  if (index === -1 || neighbour === undefined) return order;
  return moveColumn(order, id, neighbour, direction === -1 ? 'before' : 'after');
}

export interface OrderGroup {
  status: OrderStatus;
  orders: Order[];
  total: number;
}

/** Groups orders by status in the canonical STATUS_ORDER; empty statuses are skipped. */
export function groupOrdersByStatus(orders: Order[]): OrderGroup[] {
  const byStatus = new Map<OrderStatus, Order[]>();
  for (const order of orders) {
    const bucket = byStatus.get(order.status);
    if (bucket) bucket.push(order);
    else byStatus.set(order.status, [order]);
  }
  return STATUS_ORDER.filter((status) => byStatus.has(status)).map((status) => {
    const groupOrders = byStatus.get(status)!;
    return {
      status,
      orders: groupOrders,
      total: groupOrders.reduce((sum, o) => sum + o.total, 0),
    };
  });
}

/** Russian plural forms: pluralize(5, ['заказ', 'заказа', 'заказов']) → 'заказов'. */
export function pluralize(n: number, forms: [string, string, string]): string {
  const abs = Math.abs(n) % 100;
  const last = abs % 10;
  if (abs > 10 && abs < 20) return forms[2];
  if (last > 1 && last < 5) return forms[1];
  if (last === 1) return forms[0];
  return forms[2];
}

function csvCell(value: string | number): string {
  const str = String(value);
  return /[\";\\n]/.test(str) ? `\"${str.replace(/\"/g, '\"\"')}\"` : str;
}

/** Semicolon-separated CSV (Excel-friendly for ru locale). */
export function ordersToCsv(orders: Order[]): string {
  const header = ['id', 'customer', 'city', 'createdAt', 'itemsCount', 'total', 'payment', 'status'];
  const rows = orders.map((o) =>
    [o.id, o.customer, o.city, o.createdAt, o.itemsCount, o.total, o.payment, o.status].map(csvCell).join(';'),
  );
  return [header.join(';'), ...rows].join('\\n');
}
